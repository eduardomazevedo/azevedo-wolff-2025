It fixes notation, records the assumptions, and states the results in their intended
logical order.  A proof sketch follows every non-obvious statement.  The document

prose; theorem names and labels are intended to survive into the paper when possible.
\end{abstract}

\section{Model, notation, and maintained assumptions}

\subsection{Actions, output, and scores}

The agent chooses an \textbf{action} $a$ from a compact interval
$\cA\subseteq\R_+$.  Fix an interior \textbf{intended action} $a_0$.
\textbf{Output} $y$ lies in a common support $\cY\subseteq\R$ and has a strictly
positive \textbf{density or probability mass function} $f(\cdot\mid a)$ with
respect to a common \textbf{dominating measure} $\nu$.  We write expectations as
integrals with respect to $\nu$, using counting measure for discrete output.

For $\nu$-almost every $y\in\cY$, the map $a\mapsto f(y\mid a)$ is twice
continuously differentiable on an open set containing $\cA$.

The score is
\[
 S(y\mid a):=\partial_a\log f(y\mid a)=\frac{f_a(y\mid a)}{f(y\mid a)}.
\]
After fixing $a_0$, write
\[
 s(y):=S(y\mid a_0),\qquad f_0(y):=f(y\mid a_0).
\]

\subsection{Utility, wealth, payment, and cost}

The agent has \textbf{initial wealth} $w_0\in\R$ and a \textbf{utility function}
$u:[w_0,\infty)\to\R$ over final wealth.  The utility function is smooth,
strictly increasing, satisfies $u''(W)<0$ for every $W>w_0$, and
$\lim_{W\to\infty}u'(W)=0$.
Define
\[
\begin{aligned}
 \bar u&:=\lim_{x\to\infty}u(x),\\
 \underline u&:=u(w_0),\\
 \Delta u&:=\bar u-\underline u.
\end{aligned}
\]

A \textbf{contract} is a measurable function $w:\cY\to\R_+$.  Given a contract
$w$, we use two convenient reparametrizations.  The associated
\textbf{final-wealth contract} is $W(y):=w_0+w(y)$, and the associated
\textbf{utility contract} is $v(y):=u(W(y))$.  Limited liability is $w(y)\ge0$,
or equivalently $W(y)\ge w_0$.  Let $k:=u^{-1}$.

The agent incurs a smooth, strictly convex \textbf{effort cost} $c(a)$, with
$c'(a_0)>0$.  Given a payment contract, the agent's \textbf{expected utility}
and the principal's \textbf{expected compensation cost} are
\begin{align*}
 U(w,a)&:=\int u(w_0+w(y))f(y\mid a)\dd(y)-c(a),\\
 C(w,a)&:=\int w(y)f(y\mid a)\dd(y).
\end{align*}
In utility-contract coordinates, define the corresponding transformed
functionals
\begin{align*}
 \widehat U(v,a)&:=\int v(y)f(y\mid a)\dd(y)-c(a),\\
 \widehat C(v,a)&:=\int [k(v(y))-w_0]f(y\mid a)\dd(y).
\end{align*}
Thus $U(w,a)=\widehat U(v,a)$ and $C(w,a)=\widehat C(v,a)$ under the change of
variables $v=u(w_0+w)$.

The \textbf{feasible contract set} $\mathcal V$ consists of all measurable
utility contracts $v:\cY\to[\underline u,\bar u)$ that have finite expected
compensation cost $\widehat C(v,a)$ for every $a\in\cA$ and for which the map
\[
 a\longmapsto\int v(y)f(y\mid a)\dd(y)
\]
is twice continuously differentiable, with
\begin{equation}
 \frac{d^j}{da^j}\int v(y)f(y\mid a)\dd(y)
 =\int v(y)\partial_a^j f(y\mid a)\dd(y),
 \qquad j=1,2.
 \tag{Diff}\label{eq:contract-differentiation}
\end{equation}

The set $\mathcal V$ is convex.  Indeed, action regularity is preserved by
convex combinations, and convexity of $k$ implies that a convex combination of
contracts with finite expected compensation cost also has finite expected cost.

\subsection{Link functions and canonical contracts}

\begin{definition}[Marginal-cost floor and links]
The marginal cost of utility at the limited-liability floor is
\[
 m_0:=k'(\underline u)=\frac{1}{u'(w_0)}.
\]
On the nonbinding region define the wealth and utility links
\[
 L_0(z):=(u')^{-1}(1/z),
 \qquad
 \ell_0(z):=u(L_0(z))=(k')^{-1}(z).
\]
The limited-liability links are
\[
 L(z):=L_0(z\vee m_0),
 \qquad
 \ell(z):=\ell_0(z\vee m_0)=u(L(z)).
\]
Thus $L(z)=w_0$ and $\ell(z)=\underline u$ for $z\le m_0$.
At $m_0$, derivatives always mean the relevant one-sided derivative.
\end{definition}

On the nonbinding region,
\[
 u'(L_0(z))=\frac1z,
 \qquad
 \ell_0'(z)=\frac{L_0'(z)}{z}.
\]

\begin{definition}[Canonical contracts]
For multipliers $\lambda,\mu\in\R$, define
\begin{align*}
 W_{\lambda,\mu}(y)&:=L(\lambda+\mu s(y)),\\
 w_{\lambda,\mu}(y)&:=L(\lambda+\mu s(y))-w_0,\\
 v_{\lambda,\mu}(y)&:=\ell(\lambda+\mu s(y)).
\end{align*}
The first is final wealth, the second is payment, and the third is utility.
\end{definition}

We assume \textbf{canonical-contract regularity}: every canonical utility
contract $v_{\lambda,\mu}$ belongs to $\mathcal V$, and its expected utility,
local incentives, and expected compensation cost vary continuously with the
contract parameters.  Thus the maps
\[
 (\lambda,\mu)\longmapsto \widehat U(v_{\lambda,\mu},a_0),
 \qquad
 (\lambda,\mu)\longmapsto \widehat U_a(v_{\lambda,\mu},a_0),
 \qquad
 (\lambda,\mu)\longmapsto \widehat C(v_{\lambda,\mu},a_0)
\]
are continuous on $\R^2$.  Recall that membership in $\mathcal V$ already
requires finite expected compensation and valid differentiation under the
integral sign through second order.  These requirements can be verified together
by standard local domination conditions.  Distributional examples in the
appendix should verify such primitive conditions rather than build them into the
statement of the main results.

Finally, we assume \textbf{local implementability}:
\begin{equation}
 \Delta u\int_{\{s>0\}}s(y)f_0(y)\dd(y)>c'(a_0).
 \tag{Imp}\label{eq:implementability}
\end{equation}
When $\Delta u=\infty$, this means that the positive score has positive first
moment.

\section{The cost-minimization problems and the first-order approach}

For an intended action $a_0$ and reservation utility $\bar U$, the
\textbf{cost-minimization problem} minimizes expected compensation cost at the
intended action,
\[
 \widehat C(v,a_0),
\]
over contracts $v\in\mathcal V$ subject to individual rationality and global
incentive compatibility:
\begin{align}
 \widehat U(v,a_0)&\ge \bar U, \tag{IR}\label{eq:IR}\\
 \widehat U(v,a_0)&\ge \widehat U(v,a)
 \quad\text{for every }a\in\cA. \tag{GIC}\label{eq:GIC}
\end{align}
Thus GIC requires that the intended action $a_0$ be optimal for the agent under
the contract.

Because $a_0$ is interior, GIC implies the local first-order condition
$\widehat U_a(v,a_0)=0$.  The \textbf{relaxed cost-minimization problem} replaces
GIC with this necessary local condition: it minimizes $\widehat C(v,a_0)$ over
$v\in\mathcal V$ subject to
\begin{align}
 \widehat U(v,a_0)&\ge \bar U, \tag{IR}\\
 \widehat U_a(v,a_0)&=0. \tag{LIC}\label{eq:LIC}
\end{align}
By the definition of $\mathcal V$, LIC is
\[
 \int v(y)s(y)f_0(y)\dd(y)=c'(a_0).
\]

\begin{definition}[Validity of the first-order approach]
The \emph{first-order approach} is valid for $(a_0,\bar U)$ if every solution of
the relaxed cost-minimization problem is also a solution of the original
cost-minimization problem.
\end{definition}

\subsection{The LIC path}

\begin{definition}[Incentive functional]
For fixed $\lambda$ and $\mu$, let
\[
 I(\lambda,\mu):=
 \int \ell(\lambda+\mu s(y))s(y)f_0(y)\dd(y).
\]
Thus a canonical contract satisfies LIC exactly when
$I(\lambda,\mu)=c'(a_0)$.
\end{definition}

\begin{lemma}[Unique LIC multiplier]
\label{lem:unique-mu}
For every $\lambda\ge0$, there is a unique $\widetilde\mu(\lambda)>0$ such that
\[
 I(\lambda,\widetilde\mu(\lambda))=c'(a_0).
\]
The map $\lambda\mapsto\widetilde\mu(\lambda)$ is continuous.
\end{lemma}

\proofsketch
Because $\E_{a_0}[s]=0$, $I(\lambda,0)=0$.  Monotonicity of $\ell$ implies that
$I$ is weakly increasing in $\mu$; where differentiation is justified,
\[
 I_\mu(\lambda,\mu)=
 \int \ell'(\lambda+\mu s(y))s(y)^2f_0(y)\dd(y)\ge0.
\]
After subtracting the constant $\ell(\lambda)$ and using $\E_{a_0}[s]=0$,
the integrand is nonnegative and increases pointwise in $\mu$.  Hence monotone
convergence applies.  As $\mu\to\infty$,
\[
 \ell(\lambda+\mu s(y))\longrightarrow
 \begin{cases}
  \bar u,&s(y)>0,\\
  \ell(\lambda),&s(y)=0,\\
  \underline u,&s(y)<0.
 \end{cases}
\]
Consequently, with an infinite limit allowed,
\[
 I(\lambda,\mu)\longrightarrow
 \Delta u\int_{\{s>0\}}s(y)f_0(y)\dd(y)>c'(a_0).
\]
Existence follows by continuity.  At any LIC solution, positive generated
incentives imply that $\ell'$ is positive on a set with $s\ne0$, yielding strict
monotonicity at the solution and hence uniqueness.  Continuity follows from the
same strict crossing argument.

\begin{definition}[LIC path]
Define the utility contract
\[
 v_\lambda:=v_{\lambda,\widetilde\mu(\lambda)},
 \qquad
 v_\lambda(y)=\ell(\lambda+\widetilde\mu(\lambda)s(y)),
\]
with associated expected utility and cost
\begin{align*}
 \widetilde U(\lambda)&:=\widehat U(v_\lambda,a_0),\\
 \widetilde C(\lambda)&:=\widehat C(v_\lambda,a_0).
\end{align*}
\end{definition}

\subsection{Lagrangian and Pareto characterization}

\begin{lemma}[Pointwise Lagrangian minimization]
\label{lem:pointwise}
For multipliers $\lambda\ge0$ and $\mu\in\R$, consider
\[
 \mathcal L(v,\lambda,\mu)
 :=\widehat C(v,a_0)+\lambda[\bar U-\widehat U(v,a_0)]
 -\mu\widehat U_a(v,a_0).
\]
Among feasible utility contracts, its almost-everywhere unique minimizer is
$v_{\lambda,\mu}$.
\end{lemma}

\proofsketch
After dropping terms independent of $v$, the integrand is
\[
 k(v(y))-w_0-[\lambda+\mu s(y)]v(y).
\]
Strict convexity of $k$ gives a unique pointwise minimizer.  The interior first-order
condition is $k'(v)=\lambda+\mu s$; imposing $v\ge\underline u$ replaces the
right-hand side by its maximum with $m_0$, which gives $v=\ell(\lambda+\mu s)$.
If $\bar u<\infty$, the open upper endpoint causes no nonattainment: since
$k(v)\to\infty$ as $v\uparrow\bar u$, the pointwise objective also diverges at
that endpoint for every finite value of $\lambda+\mu s(y)$.

\begin{lemma}[Relaxed Pareto path]
\label{lem:pareto-path}
For every $\lambda\ge0$, $v_\lambda$ is the almost-everywhere
unique minimizer of
\[
 \widehat C(v,a_0)-\lambda\widehat U(v,a_0)
\]
among contracts satisfying LIC.  The function $\widetilde U$ is continuous and
nondecreasing.  Moreover, if $\lambda_1<\lambda_2$ and
$\widetilde U(\lambda_1)=\widetilde U(\lambda_2)$, then
$v_{\lambda_1}=v_{\lambda_2}$ almost everywhere.
\end{lemma}

\proofsketch
The first claim follows from Lemma~\ref{lem:pointwise} with
$\mu=\widetilde\mu(\lambda)$.  For $\lambda_1<\lambda_2$, the two Pareto inequalities
imply
\[
 (\lambda_2-\lambda_1)
 [\widetilde U(\lambda_2)-\widetilde U(\lambda_1)]\ge0.
\]
If the two utility levels are equal, both contracts solve the same strictly
convex Pareto problem, so uniqueness forces them to agree almost everywhere.
Continuity follows from continuity of the canonical path.  Thus a flat portion
of $\widetilde U$ represents one economic contract supported by multiple
multipliers; geometrically, it corresponds to a corner of the cost--utility
frontier.

\begin{definition}[Reservation-utility frontiers]
Define
\[
 U_L:=\widetilde U(0),
 \qquad
 U_R:=\lim_{\lambda\to\infty}\widetilde U(\lambda)
 \in\R\cup\{\infty\}.
\]
The limit exists because $\widetilde U$ is nondecreasing.
\end{definition}

\begin{proposition}[Generalized relaxed solution]
\label{prop:relaxed-solution}
The frontiers satisfy $U_L<U_R$.  For each $\bar U<U_R$, let
$\Lambda(\bar U)$ denote the set of valid IR multipliers $\lambda$ for which
$v_\lambda$ represents the relaxed solution.  Then:
\begin{enumerate}[label=(\roman*)]
 \item the relaxed problem has an almost-everywhere unique solution,
 $\Lambda(\bar U)$ is a nonempty compact interval, and, for every
 $\lambda\in\Lambda(\bar U)$,
 \[
  v^*_{\bar U}(y)
  =v_\lambda(y)
  =\ell\bigl(\lambda+\widetilde\mu(\lambda)s(y)\bigr);
 \]
 thus the contract is unique although its multipliers need not be;
 \item if $\bar U\le U_L$, the relaxed contract and its expected cost are
 constant in $\bar U$, and $\Lambda(\bar U)=\{0\}$ when $\bar U<U_L$; if
 $\bar U\in(U_L,U_R)$, IR binds.  On this active region, $\Lambda$ is strictly
 increasing: every multiplier associated with a higher reservation utility
 exceeds every multiplier associated with a lower one.  Moreover,
 $\inf\Lambda(\bar U)$ converges to infinity as $\bar U\to U_R$;
 \item the relaxed value $C^*(\bar U)$ is constant below $U_L$ and strictly
 increasing and strictly convex on $(U_L,U_R)$.  On this active region its
 supporting slopes are exactly the IR multipliers.
\end{enumerate}
\end{proposition}

\proofsketch
Continuity and monotonicity of $\widetilde U$ imply that every utility level in
$(U_L,U_R)$ is attained.  Any two multipliers attaining the same utility induce
the same contract by Lemma~\ref{lem:pareto-path}.  The supporting-multiplier
argument then gives optimality and uniqueness, with binding IR above $U_L$.
Below $U_L$, the $\lambda=0$ Pareto contract is feasible and minimizes cost, and
complementary slackness makes it the only valid IR multiplier.  The sets
$\Lambda(\bar U)$ are intervals because they are level sets of the continuous
nondecreasing function $\widetilde U$ (with the complementary-slackness
restriction below $U_L$); closedness follows from continuity, and boundedness
follows from the no-terminal-flat argument below.

Strict convexity of the value follows directly in utility-contract space.  If
$\bar U_1\ne\bar U_2$ are in $(U_L,U_R)$, IR binds at their distinct optimal
contracts $v_1$ and $v_2$.  A strict convex combination of these contracts
satisfies LIC and delivers the corresponding convex combination of utilities,
while strict convexity of $k$ makes its expected compensation cost strictly less
than the same convex combination of costs.  Strict increase follows from binding
IR and the supporting-multiplier characterization below.

Fix $\bar U\in(U_L,U_R)$ and $\lambda\in\Lambda(\bar U)$.  Pareto optimality of
$v_\lambda$ implies, for every other active reservation utility $T$,
\[
 C^*(T)\ge C^*(\bar U)+\lambda(T-\bar U),
\]
with the same inequality below $U_L$ because the value is constant there and
$\lambda\ge0$.  Hence $\lambda\in\partial C^*(\bar U)$.  Conversely, take
$\lambda\in\partial C^*(\bar U)$ and put $T=\widetilde U(\lambda)$.  The
subgradient inequality and Pareto optimality give opposite inequalities between
$C^*(T)-\lambda T$ and $C^*(\bar U)-\lambda\bar U$, so equality holds.  If
$T\ne\bar U$, the same supporting line touches $C^*$ at two distinct utility
levels, forcing the value to be affine between them and contradicting strict
convexity.  Thus $T=\bar U$ and $\lambda\in\Lambda(\bar U)$.  This proves
$\Lambda(\bar U)=\partial C^*(\bar U)$.  Standard one-dimensional convexity then
makes this set the compact interval between the one-sided derivatives.  If
$\bar U_1<\bar U_2$, subgradient monotonicity weakly orders their multiplier
sets; equality of any two slopes would again make $C^*$ affine between the two
points, so the ordering is strict.

It remains to rule out a terminal flat portion of $\widetilde U$.  If one
existed, Lemma~\ref{lem:pareto-path} would imply that the same contract is
canonical for arbitrarily large $\lambda$.  LIC and $c'(a_0)>0$ imply that this
contract is above the limited-liability floor on a positive-score set.  At any
such outcome the interior condition gives
$k'(v(y))=\lambda+\widetilde\mu(\lambda)s(y)>\lambda$, which is impossible for a
fixed contract as $\lambda\to\infty$.  Hence
$\widetilde U(M)<U_R$ for every finite $M$.  Monotonicity now implies that if
$\bar U>\widetilde U(M)$, every $\lambda\in\Lambda(\bar U)$ exceeds $M$.
Because $M$ is arbitrary, $\inf\Lambda(\bar U)\to\infty$.

\section{Score-tail geometry and the limited-liability kink}

Write
\[
 \underline c'':=\inf_{a\in\cA}c''(a)\ge0.
\]
Conditions requiring uniformly positive cost curvature will state
$\underline c''>0$ explicitly.  For utility with $u''(W)<0$ on
$(w_0,\infty)$, define absolute risk aversion and absolute prudence by
\[
 A(W):=-\frac{u''(W)}{u'(W)},
 \qquad
 P(W):=-\frac{u'''(W)}{u''(W)}.
\]

\begin{assumption}[Prudence--risk aversion bound]
\label{ass:prudence-risk-aversion}
For every $W>w_0$,
\[
 P(W)\le3A(W).
\]
\end{assumption}

\begin{assumption}[Vanishing marginal utility relative to risk aversion]
\label{ass:vanishing-marginal-utility}
Marginal utility vanishes relative to absolute risk aversion:
\[
 \frac{u'(W)}{A(W)}\longrightarrow0
 \qquad\text{as }W\to\infty.
\]
\end{assumption}

\begin{remark}[Link interpretation]
Assumption~\ref{ass:prudence-risk-aversion} is equivalent to concavity of the
nonbinding utility link $\ell_0$; monotonicity of $\ell_0$ follows from the
maintained utility assumptions.  Moreover, at $W=L_0(z)$,
\[
 L_0'(z)
 =\frac{u'(W)}{A(W)}
 =\frac{[u'(W)]^2}{-u''(W)},
 \qquad
 z\ell_0'(z)=L_0'(z).
\]
Thus Assumption~\ref{ass:vanishing-marginal-utility} is equivalent both to an
asymptotically flat wealth link, $L_0'(z)\to0$, and to
$z\ell_0'(z)\to0$.
\end{remark}

\begin{definition}[Limited-liability score cutoff]
For $\mu>0$, define
\[
 q^{\mathrm{LL}}(\lambda,\mu):=\frac{m_0-\lambda}{\mu}.
\]
Limited liability binds precisely on
\[
 \{y:s(y)\le q^{\mathrm{LL}}(\lambda,\mu)\}.
\]
Along the LIC path write
\[
 \widetilde q^{\mathrm{LL}}(\lambda)
 :=q^{\mathrm{LL}}(\lambda,\widetilde\mu(\lambda)).
\]
\end{definition}

\begin{definition}[Tail-safe cutoffs]
A number $q\le0$ is \emph{tail-safe} if
\[
 f_{aa}(y\mid a)\ge0
 \quad\text{for every }a\in\cA
 \text{ and every }y\text{ with }s(y)\le q.
 \tag{TS}\label{eq:tail-safe}
\]
Let
\[
 \cQ_{\mathrm{safe}}:=\{q\le0:q\text{ is tail-safe}\}.
\]
For the cutoff-specific arguments in the proofs, define incentive capacity and
paid-region positive curvature by
\begin{align*}
 \Capacity(q)&:=-\int_{\{s(y)\le q\}}s(y)f_0(y)\dd(y),\\
 \Curv(q)&:=\sup_{a\in\cA}
 \int_{\{s(y)>q\}}[s(y)f_{aa}(y\mid a)]_+\dd(y).
\end{align*}
The set $\cQ_{\mathrm{safe}}$ is a lower interval.  Define its right endpoint,
safe incentive capacity, and best paid-region curvature bound by
\begin{align*}
 \bar q_{\mathrm{safe}}&:=\sup\cQ_{\mathrm{safe}},\\
 \Capacity_{\mathrm{safe}}
 &:=-\int_{\{s(y)<\bar q_{\mathrm{safe}}\}}s(y)f_0(y)\dd(y),\\
 \Curv_{\mathrm{safe}}
 &:=\inf_{q\in\cQ_{\mathrm{safe}}}\sup_{a\in\cA}
 \int_{\{s(y)>q\}}[s(y)f_{aa}(y\mid a)]_+\dd(y).
\end{align*}
In the manuscript body, define only the aggregate quantities
$\Capacity_{\mathrm{safe}}$ and $\Curv_{\mathrm{safe}}$, writing their integral
definitions directly.  The cutoff-specific notation $\Capacity(q)$ and
$\Curv(q)$ is proof machinery and can be introduced later.  For the logarithmic
result, also define
\[
 \underline s:=\operatorname*{ess\,inf}_{Y\sim f_0}s(Y),
 \qquad
 \Width_{\mathrm{safe}}:=
 \begin{cases}
  \bar q_{\mathrm{safe}}-\underline s,
   &\cQ_{\mathrm{safe}}\ne\varnothing,\\
  0,&\cQ_{\mathrm{safe}}=\varnothing.
 \end{cases}
\]
When the curvature--width ratio appears below, a finite numerator divided by an
infinite width is interpreted as zero; the ratio is infinity if its numerator is
infinite or its denominator is zero.
\end{definition}

Throughout, when $\Delta u=\infty$, a capacity condition of the form
$\Delta u\,K>c'(a_0)$ means $K>0$.  This convention applies both to
$K=\Capacity(q)$ and to $K=\Capacity_{\mathrm{safe}}$.

The weak inequality in $\{s(y)\le q\}$ assigns a score atom at $q$ to the safe,
no-payment side.  Its strict complement appears in $\Curv(q)$.  The function
$\Capacity(q)$ is increasing in $q$, whereas $\Curv(q)$ is decreasing.  Safe
incentive capacity excludes any capacity concentrated at a rightmost safe score
atom with no larger safe cutoff.

\section{Finite-\texorpdfstring{$\lambda$}{lambda} concavity lemmas}

\begin{definition}[Trial multiplier]
For $\lambda>m_0$ and $q<0$, let
\[
 \mu_q(\lambda):=\frac{m_0-\lambda}{q}>0.
\]
This multiplier places the kink at $q$:
$\lambda+\mu_q(\lambda)q=m_0$.
\end{definition}

\begin{lemma}[Kink placement]
\label{lem:kink-placement}
Fix $q<0$.  If
\[
 \Delta u\,\Capacity(q)>c'(a_0),
\]
then, for all sufficiently large $\lambda$,
\[
 \widetilde q^{\mathrm{LL}}(\lambda)\le q,
 \qquad
 \widetilde\mu(\lambda)\le\mu_q(\lambda)=O(\lambda).
\]
Tail safety of $q$ is not needed for this lemma, although later applications use
a tail-safe $q$.
\end{lemma}

\proofsketch
At $\mu_q(\lambda)$, scores below $q$ receive $\underline u$.  Subtracting the
constant $\ell(\lambda)$ and using $\E_{a_0}[s]=0$ gives
\[
 I(\lambda,\mu_q(\lambda))
 \ge [\ell(\lambda)-\underline u]\Capacity(q).
\]
All omitted terms are nonnegative: for negative scores above $q$, both the utility
difference and the score are nonpositive; for positive scores both are nonnegative.
The strict capacity condition makes the right side exceed $c'(a_0)$ eventually.
Monotonicity of $I$ then gives $\widetilde\mu\le\mu_q$.  Since $m_0-\lambda<0$,
dividing by the smaller positive multiplier moves the cutoff weakly left:
$\widetilde q^{\mathrm{LL}}(\lambda)\le q$.

\begin{definition}[Secant slope]
Along the LIC path, abbreviate $\widetilde\mu=\widetilde\mu(\lambda)$ and define
\[
 D_\lambda\ell(y):=
 \frac{\ell(\lambda+\widetilde\mu s(y))-\ell(\lambda)}{\widetilde\mu s(y)}
\]
when $s(y)\ne0$, with its continuous extension when $s(y)=0$.
\end{definition}

\begin{lemma}[Curvature identity]
\label{lem:curvature-identity}
For every $a\in\cA$,
\begin{equation}
 \widehat U_{aa}(v_\lambda,a)
 =\widetilde\mu
 \int D_\lambda\ell(y)s(y)f_{aa}(y\mid a)\dd(y)-c''(a).
 \tag{Curv}\label{eq:curvature-identity}
\end{equation}
\end{lemma}

\proofsketch
Add and subtract the constant $\ell(\lambda)$ from the utility contract and use
$\int f_{aa}(y\mid a)\dd(y)=0$.  Differentiation under the integral sign then
gives the result.

\begin{lemma}[Curvature split]
\label{lem:curvature-split}
Suppose Assumption~\ref{ass:prudence-risk-aversion} holds.  Fix
$q\in\cQ_{\mathrm{safe}}$ with $\Curv(q)<\infty$.  If the limited-liability cutoff
is weakly below $q$,
\[
 \widetilde q^{\mathrm{LL}}(\lambda)\le q,
\]
then, for every $a\in\cA$,
\begin{equation}
 \widehat U_{aa}(v_\lambda,a)
 \le
 \widetilde\mu\,\ell_0'(\lambda+\widetilde\mu q)\Curv(q)-c''(a).
 \tag{Split}\label{eq:curvature-split}
\end{equation}
If the argument equals $m_0$, use the right derivative.
\end{lemma}

\proofsketch
On $\{s(y)\le q\}$, tail safety gives $f_{aa}\ge0$, while $s\le0$ and monotonicity gives
$D_\lambda\ell\ge0$.  Hence this region contributes weakly negatively to
\eqref{eq:curvature-identity}.  On $\{s>q\}$ the contract is nonbinding.
Concavity of $\ell_0$ bounds every relevant secant slope by
$\ell_0'(\lambda+\widetilde\mu q)$.  Retaining only the positive part of
$s f_{aa}$ and taking the supremum over actions gives \eqref{eq:curvature-split}.

\begin{proposition}[Finite-$\lambda$ safe-tail curvature bound]
\label{prop:deterministic-criterion}
Suppose Assumption~\ref{ass:prudence-risk-aversion} holds.  Fix
$q_0<q_1\le0$ with $q_0,q_1\in\cQ_{\mathrm{safe}}$.  Suppose $q_0$
has sufficient incentive capacity and paid-region curvature is finite at $q_1$:
\[
 \Delta u\,\Capacity(q_0)>c'(a_0),
 \qquad
 \Curv(q_1)<\infty.
\]
For all sufficiently large $\lambda$,
\begin{equation}
 \sup_{a\in\cA}\widehat U_{aa}(v_\lambda,a)
 \le
 \mu_{q_0}(\lambda)
 \ell_0'(\lambda+\mu_{q_0}(\lambda)q_1)\Curv(q_1)-\underline c''.
 \tag{DC-$\lambda$}\label{eq:deterministic-finite}
\end{equation}
\end{proposition}

\proofsketch
Kink placement at $q_0$ gives
$\widetilde q^{\mathrm{LL}}(\lambda)\le q_0<q_1$ and
$\widetilde\mu\le\mu_{q_0}$.  For $q_1\le0$, the map
\[
 \mu\longmapsto \mu\ell_0'(\lambda+\mu q_1)
\]
is nondecreasing on the nonbinding range: increasing $\mu$ directly increases
the first factor and weakly lowers the argument of the decreasing derivative.
Combine this observation with Lemma~\ref{lem:curvature-split} at $q_1$ and
$c''(a)\ge\underline c''$.

\section{Asymptotic verification and the main theorem}

Proposition~\ref{prop:deterministic-criterion} gives the general asymptotic
criterion used below.  If there are $q_0<q_1\le0$ in
$\cQ_{\mathrm{safe}}$ such that
\[
 \Delta u\,\Capacity(q_0)>c'(a_0)
\]
and
\begin{equation}
 \limsup_{\lambda\to\infty}
 \mu_{q_0}(\lambda)
 \ell_0'(\lambda+\mu_{q_0}(\lambda)q_1)\Curv(q_1)
 <\underline c'',
 \tag{DC}\label{eq:DC}
\end{equation}
then the LIC-path contracts induce an agent objective concave in action for all
sufficiently large $\lambda$.  Because LIC holds at the interior action $a_0$,
these contracts also satisfy global incentive compatibility.  This criterion is
proof machinery rather than a headline result; the next two lemmas verify it
using aggregate primitives.

\subsection{Vanishing marginal utility relative to risk aversion}

\begin{lemma}[Vanishing-marginal-utility verification]
\label{lem:vanishing-marginal-utility}
Suppose Assumptions~\ref{ass:prudence-risk-aversion} and
\ref{ass:vanishing-marginal-utility} hold, $\underline c''>0$, and
\[
 \Delta u\,\Capacity_{\mathrm{safe}}>c'(a_0),
 \qquad
 \Curv_{\mathrm{safe}}<\infty.
\]
Then the LIC-path contracts induce an agent objective concave in action for all
sufficiently large $\lambda$.
\end{lemma}

\proofsketch
The aggregate conditions provide $q_0<q_1$ in $\cQ_{\mathrm{safe}}$ such that
$\Delta u\,\Capacity(q_0)>c'(a_0)$ and $\Curv(q_1)<\infty$.  The trial
multiplier is $\mu_{q_0}(\lambda)=O(\lambda)$, while
\[
 z_{01}(\lambda):=\lambda+\mu_{q_0}(\lambda)q_1
 \sim \lambda\left(1-\frac{q_1}{q_0}\right).
\]
Because $q_0<q_1$, the coefficient is positive.  Therefore
\[
 \mu_{q_0}(\lambda)\ell_0'(z_{01}(\lambda))\Curv(q_1)\longrightarrow0
\]
by $z\ell_0'(z)\to0$.  Thus \eqref{eq:DC} holds.

\subsection{The logarithmic borderline}

For log utility, $\ell_0'(z)=1/z$, so $z\ell_0'(z)=1$ and marginal utility does
not vanish relative to risk aversion.

\begin{lemma}[Log-width verification]
\label{lem:log-width}
Suppose $u(W)=\log W$ and
\begin{equation}
 \frac{\Curv_{\mathrm{safe}}}{\Width_{\mathrm{safe}}}
 <\underline c''.
 \tag{LogWidth}\label{eq:log-width}
\end{equation}
Then the LIC-path contracts induce an agent objective concave in action for all
sufficiently large $\lambda$.
\end{lemma}

\proofsketch
Log utility satisfies Assumption~\ref{ass:prudence-risk-aversion}.  The strict
endpoint inequality provides
$\underline s<q_0<q_1\le0$ with $q_0,q_1\in\cQ_{\mathrm{safe}}$ such that
$\Capacity(q_0)>0$ and
\[
 \frac{\Curv(q_1)}{q_1-q_0}<\underline c''.
\]
Log utility is unbounded, so the first condition gives capacity.  Moreover,
\[
 \mu_{q_0}(\lambda)
 \ell_0'(\lambda+\mu_{q_0}(\lambda)q_1)
 =\frac{\mu_{q_0}(\lambda)}
 {\lambda+\mu_{q_0}(\lambda)q_1}
 \longrightarrow\frac1{q_1-q_0}.
\]
Thus \eqref{eq:DC} holds.

\subsection{Passage to reservation utility and the FOA}

\begin{theorem}[FOA at high feasible reservation utility]
\label{thm:high-reservation}
Suppose either
\begin{enumerate}[label=(\roman*)]
 \item Utility satisfies Assumptions~\ref{ass:prudence-risk-aversion} and
 \ref{ass:vanishing-marginal-utility}, and
 \[
  \underline c''>0,
  \qquad
  \Delta u\,\Capacity_{\mathrm{safe}}>c'(a_0),
  \qquad
  \Curv_{\mathrm{safe}}<\infty;
 \]
 or
 \item Utility is logarithmic and
 \[
  \frac{\Curv_{\mathrm{safe}}}{\Width_{\mathrm{safe}}}
  <\underline c''.
 \]
\end{enumerate}
Then there exists $U^*<U_R$ such that the first-order approach is valid for every
\[
 \bar U\in[U^*,U_R).
\]
For every such $\bar U$, the original cost-minimization problem has the almost
everywhere unique solution characterized by Proposition~\ref{prop:relaxed-solution};
in particular, for some associated multipliers $\lambda\ge0$ and $\mu>0$,
\[
 v^*_{\bar U}(y)=\ell\bigl(\lambda+\mu s(y)\bigr).
\]
\end{theorem}

\proofsketch
Lemma~\ref{lem:vanishing-marginal-utility} or Lemma~\ref{lem:log-width} supplies
$\lambda_0$.  Proposition~\ref{prop:relaxed-solution} implies that, sufficiently
close to $U_R$, every canonical representation of the relaxed-optimal contract
has $\lambda\ge\lambda_0$.  Choose $U^*$ so that this holds above $U^*$.
Concavity and LIC at the interior action $a_0$ give global incentive
compatibility.  Thus the unique relaxed solution is feasible for, and uniquely
solves, the original problem.

\section{Utility corollaries}

The following statements hold the distribution, intended action, effort cost,
and utility-shape parameter fixed.  Parts~(i) and (ii) compare a family of models
indexed only by initial wealth.  At each $w_0$ covered by the result, the
regularity conditions for the relaxed solution are understood to hold, and the
frontier $U_R(w_0)$ and threshold $U^*(w_0)$ are recomputed for that model.  No
reservation-utility threshold uniform in $w_0$ is claimed.

\begin{corollary}[Common utility specifications]
\label{cor:common-utilities}
Under the maintained conditions and $\underline c''>0$, the first-order
approach is valid as follows:
\begin{enumerate}[label=(\roman*)]
 \item Suppose $\Capacity_{\mathrm{safe}}>0$ and
 $\Curv_{\mathrm{safe}}<\infty$.  For fixed $\alpha>0$ and CARA utility
 $u(W)=-e^{-\alpha W}/\alpha$, there exists $w_0^\dagger\in\R$ such that, for
 every $w_0\le w_0^\dagger$, there exists
 $U^*(w_0)<U_R(w_0)$ and the FOA is valid for every
 \[
  \bar U\in[U^*(w_0),U_R(w_0)).
 \]
 \item Suppose $\Capacity_{\mathrm{safe}}>0$ and
 $\Curv_{\mathrm{safe}}<\infty$.  For fixed $\gamma>1$ and CRRA utility
 $u(W)=W^{1-\gamma}/(1-\gamma)$, there exists $w_0^\dagger>0$ such that, for
 every $0<w_0\le w_0^\dagger$, there exists
 $U^*(w_0)<U_R(w_0)$ and the FOA is valid for every
 \[
  \bar U\in[U^*(w_0),U_R(w_0)).
 \]
 \item If $w_0>0$, $u(W)=\log W$, $\underline s=-\infty$, and
 $\Curv_{\mathrm{safe}}<\infty$, then there is $U^*<\infty$ such that validity
 holds for every reservation utility $\bar U\ge U^*$.
\end{enumerate}
\end{corollary}

\proofsketch
CARA and CRRA utility with $\gamma>1$ satisfy both utility assumptions, while their
utility spreads diverge as $w_0\to-\infty$ and $w_0\downarrow0$, respectively.
Because $a_0$, the distribution, and effort cost are fixed,
$\Capacity_{\mathrm{safe}}$, $\Curv_{\mathrm{safe}}$, and $c'(a_0)$ do not vary
with $w_0$.  Choose $w_0^\dagger$ so that the safe-capacity inequality in
Theorem~\ref{thm:high-reservation} holds throughout the stated range.  Applying
the theorem separately at each such $w_0$ supplies the possibly different
thresholds $U^*(w_0)$.  For log utility, the safe score width and $U_R$ are both
infinite.

\section{Distribution verifications}

These calculations are intended for an appendix.  In the manuscript's
closed-form-solutions subsection, the distribution table should contain only a
brief note that the named examples other than Student's $t$ satisfy the relevant
qualitative safe-tail and integrability conditions under the stated parameter
restrictions, with verification deferred to the appendix.

\subsection{Gaussian location}

Suppose $Y\sim N(a,\sigma^2)$.  Then
\[
 s(y)=\frac{y-a_0}{\sigma^2},
 \qquad
 f_{aa}(y\mid a)=f(y\mid a)
 \left[\frac{(y-a)^2}{\sigma^4}-\frac1{\sigma^2}\right].
\]
If $\cA=[\underline a,\bar a]$, the right edge of the uniformly safe low-score
region is
\[
 \bar q_{\mathrm{safe}}
 =\min\left\{0,\frac{\underline a-a_0-\sigma}{\sigma^2}\right\}.
\]
Under $a_0$, $s(Y)\sim N(0,1/\sigma^2)$, and
\[
 \Capacity(q)=\frac1\sigma\varphi(\sigma q).
\]
Paid-region curvature is finite on compact action sets, so
$\Curv_{\mathrm{safe}}<\infty$.  Since the score is unbounded below, the log-width
condition follows.  For CARA and CRRA with $\gamma>1$, safe incentive capacity
remains a substantive condition.

\subsection{Poisson}

Suppose $Y\sim\operatorname{Pois}(a)$ and $\underline a:=\min\cA>0$.  Then
\[
 s(y)=\frac{y-a_0}{a_0},
 \qquad
 p_{aa}(y\mid a)=p(y\mid a)\frac{(y-a)^2-y}{a^2}.
\]
A count $y$ is uniformly low-tail safe if
\[
 y+\sqrt y\le\underline a.
\]
For a safe integer cutoff $K<a_0$, let $q_K:=(K-a_0)/a_0$ be the
corresponding score cutoff.  Then
\[
 \Capacity(q_K)
 =-\sum_{y\le K}\frac{y-a_0}{a_0}p(y\mid a_0)
 =\Pr_{a_0}(Y=K).
\]
Paid-region curvature is finite on compact action sets bounded away from zero.
Because score values are discrete, formal applications should compute interior
safe capacity rather than mechanically using capacity at the rightmost safe
atom.

\subsection{Exponential output with mean action}

Suppose output has density
\[
 f(y\mid a)=\frac1a e^{-y/a}\ind\{y\ge0\},
 \qquad \underline a:=\min\cA>0.
\]
Then
\[
 s(y)=\frac{y-a_0}{a_0^2},
 \qquad
 f_{aa}(y\mid a)=f(y\mid a)
 \frac{y^2-4ay+2a^2}{a^4}.
\]
The uniformly safe low-output region is
\[
 y\le(2-\sqrt2)\underline a,
\]
so
\[
 \bar q_{\mathrm{safe}}
 =\min\left\{0,
 \frac{(2-\sqrt2)\underline a-a_0}{a_0^2}
 \right\}.
\]
For a nonnegative output cutoff $r=a_0+a_0^2q$ and $b=r/a_0$,
\[
 \Capacity(q)=\frac{be^{-b}}{a_0}.
\]
Again, $\Curv(q)<\infty$ on compact action sets bounded away from zero.

\subsection{Other distributions in the solutions table}

The appendix should also record short verifications for the lognormal,
Bernoulli, geometric, binomial, and gamma examples.  Lognormal follows from the
Gaussian calculation after the logarithmic transformation.  The discrete
families are handled with score cutoffs and sums, as in the Poisson calculation,
and require action sets bounded away from parameter boundaries where relevant.
Generic exponential, location, and scale families should be accompanied by the
primitive restrictions under which the qualitative conditions hold rather than
claimed without qualification.  Student's $t$ is the deliberate exception: its
nonmonotone score does not generate the required safe low-score geometry in the
counterexample and should remain in the assumptions-and-counterexamples
subsection.

\section{Appendix-only asymptotic-affinity extension}

This section uses a different mechanism from safe-tail flattening.  It is intended
for an appendix, with at most a brief main-text reference.

\subsection{Affine-score curvature and primitive moments}

\begin{assumption}[Affine-score curvature]
\label{ass:affine-score}
For every $a\in\cA$,
\begin{equation}
 \int s(y)f_{aa}(y\mid a)\dd(y)\le0.
 \tag{Affine}\label{eq:affine-score}
\end{equation}
Equivalently, $a\mapsto\E_a[s(Y)]$ is concave.
\end{assumption}

\begin{assumption}[Moments for asymptotic affinity]
\label{ass:affinity-moments}
For some $\eta>0$,
\begin{align*}
 0<\sigma_s^2&:=\int s(y)^2f_0(y)\dd(y)<\infty,\\
 \int |s(y)|^{2+\eta}f_0(y)\dd(y)&<\infty,\\
 \sup_{a\in\cA}\int
 (1+|s(y)|^{2+\eta})|f_{aa}(y\mid a)|\dd(y)&<\infty.
\end{align*}
\end{assumption}

\subsection{CRRA with one-half below gamma below one}

Suppose
\[
 \frac12<\gamma<1,
 \qquad
 p:=\frac{1-\gamma}{\gamma}\in(0,1).
\]
On the nonbinding region,
\[
 \ell_0(z)=\frac{z^p}{1-\gamma},
 \qquad
 \ell_0'(z)=\frac1\gamma z^{p-1}.
\]
Thus $\ell_0'(z)\to0$ but $z\ell_0'(z)\to\infty$, so marginal utility does not
vanish relative to risk aversion.

\begin{lemma}[LIC multiplier asymptotics]
\label{lem:affinity-multiplier}
Under Assumption~\ref{ass:affinity-moments}, for CRRA with
$\gamma\in(1/2,1)$,
\[
 \widetilde\mu(\lambda)\ell_0'(\lambda)
 \longrightarrow
 \theta^*:=\frac{c'(a_0)}{\sigma_s^2},
 \qquad
 \frac{\widetilde\mu(\lambda)}{\lambda}\longrightarrow0.
\]
\end{lemma}

\proofsketch
For bounded $\theta$, set $\mu_\lambda(\theta)=\theta/\ell_0'(\lambda)$.  Since
$\lambda\ell_0'(\lambda)\to\infty$, we have $\mu_\lambda(\theta)/\lambda\to0$.
A central-region Taylor expansion and the $(2+\eta)$ score moment give
\[
 \ell(\lambda+\mu_\lambda(\theta)s)
 =\ell(\lambda)+\theta s+o(1)
\]
in the weighted space
\[
 L^1\bigl(|s(y)|f_0(y)\dd(y)\bigr).
\]
The constant vanishes after multiplication by $s$ and integration.  Hence
$I(\lambda,\mu_\lambda(\theta))\to\theta\sigma_s^2$.
Monotonicity of $I$ pins down the LIC solution.  The second limit follows by
dividing the first by $\lambda\ell_0'(\lambda)$.

\begin{lemma}[Uniform residual]
\label{lem:uniform-residual}
Under Assumption~\ref{ass:affinity-moments},
\begin{equation}
\begin{aligned}
 &\sup_{a\in\cA}\Bigg|
 \int\left[
  \ell(\lambda+\widetilde\mu(\lambda)s(y))-\ell(\lambda)
  -\widetilde\mu(\lambda)\ell_0'(\lambda)s(y)
 \right]
 \\[-2pt]
 &\hspace{5cm}{}
 \times f_{aa}(y\mid a)\dd(y)
 \Bigg|\longrightarrow0.
\end{aligned}
\tag{R}\label{eq:residual}
\end{equation}
\end{lemma}

\proofsketch
Split at $|s|\le \lambda/(2\widetilde\mu)$.  On the central region, Taylor's theorem
bounds the residual by a constant times
$\widetilde\mu^2|\ell_0''(\lambda/2)|s^2$, whose coefficient is $O(\lambda^{-p})$.
On the tail, use the sublinear growth $|\ell(z)|\le C(1+|z|^p)$ and the uniform
$(2+\eta)$ weighted moment of $f_{aa}$.  Lemma~\ref{lem:affinity-multiplier}
gives the rates needed for both pieces to vanish uniformly in $a$.

\begin{proposition}[Asymptotic-affinity extension]
\label{prop:asymptotic-affinity}
Suppose $\underline c''>0$, Assumptions~\ref{ass:affine-score} and
\ref{ass:affinity-moments} hold, and utility is CRRA with
$\gamma\in(1/2,1)$.  Then
\[
 \sup_{a\in\cA}
 \widehat U_{aa}(v_\lambda,a)
 \le-\underline c''+o(1).
\]
In particular, the agent's objective is concave for all sufficiently large
$\lambda$, and the FOA is valid at all sufficiently high feasible reservation
utilities.
\end{proposition}

\proofsketch
Use $\int f_{aa}\dd=0$ to remove $\ell(\lambda)$ and expand
\begin{align*}
 \int \ell(\lambda+\widetilde\mu s)f_{aa}(y\mid a)\dd(y)
 &=\widetilde\mu\ell_0'(\lambda)
   \int s(y)f_{aa}(y\mid a)\dd(y)+o(1).
\end{align*}
The remainder is uniform by Lemma~\ref{lem:uniform-residual}; the leading term is
nonpositive by affine-score curvature.  Subtract $c''(a)\ge\underline c''$.

\begin{corollary}[Gaussian CRRA extension]
For a Gaussian location family on a compact action set, all moments in
Assumption~\ref{ass:affinity-moments} hold and
\[
 \int s(y)f_{aa}(y\mid a)\dd(y)
 =\frac{d^2}{da^2}\E_a[s(Y)]=0.
\]
Consequently, Gaussian output with CRRA utility is covered by the
vanishing-marginal-utility branch of Theorem~\ref{thm:high-reservation} for
$\gamma>1$,
by its log branch for $\gamma=1$, and by asymptotic affinity for
$\gamma\in(1/2,1)$, subject in each branch to the assumptions of the corresponding
relaxed-solution result.
\end{corollary}

No general claim is made here for $\gamma\le1/2$.
