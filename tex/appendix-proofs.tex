\subsection{Notation and Utility Parametrization}
\label{subsec:appendix-notation}

In the appendix we use the following additional notation. Let
\[
    \underline u:=u(w_0),\qquad
    \bar u:=\lim_{W\to\infty}u(W),\qquad
    \Delta u:=\bar u-\underline u,
\]
and $k:=u^{-1}$.

It will be convenient to parametrize contracts in utility units because this makes the relaxed problem
a convex program with linear constraints. Given a (payment) contract $w$, define the \textbf{utility contract} parametrization by
$v(y):=u(w_0+w(y))$.  Conversely, a utility contract $v$ corresponds to the
payment contract $w(y) = k(v)-w_0$. Define
\begin{align*}
    \widehat U(v,a)&:=\int v(y)f(y\mid a)\,d\nu(y)-c(a),\\
    \widehat C(v,a)&:=\int [k(v(y))-w_0]f(y\mid a)\,d\nu(y).
\end{align*}
Local incentive compatibility in utility units is
\begin{equation}
    \widehat U_a(v,a_0)
    =0.
    \tag{$\widehat{\mathrm{LIC}}$}\label{LIC_hat}
\end{equation}
The Lagrangian in utility units is
\begin{equation}
    \widehat{\mathcal L}(v,\lambda,\mu)
    :=\widehat C(v,a_0)
      +\lambda[\bar U-\widehat U(v,a_0)]
      -\mu\widehat U_a(v,a_0).
    \label{eq:utility-lagrangian}
\end{equation}

Define the feasible-contract set in utility units by
\[
    \mathcal V
    :=\left\{v:\mathcal Y\to[\underline u,\bar u):
    k(v)-w_0\in\mathcal W\right\}.
\]
$\mathcal V$ is convex: differentiation is preserved by convex combinations, and
convexity of $k$ preserves finiteness of expected compensation cost. In the remainder of the appendix, we work with this equivalent utility parametrization of contracts and refer to utility contracts simply as contracts.

Define the utility links
\begin{eqnarray*}
    \ell_0(z)&:=&u(L_0(z))=(k')^{-1}(z),\\
    \ell(z)&:=&\ell_0(z\vee m_0).
\end{eqnarray*}
The utility parametrization of a
canonical contract is
\[
    v_{\lambda,\mu}(y):=\ell(\lambda+\mu s(y)).
\]


\subsection{Solution of the relaxed problem}
\label{subsec:appendix-relaxed}

We now prove the characterization Proposition \ref{prop:relaxed-optimal-contract}. This is a standard argument made in similar models \parencite{holmstrom1978incentives,jewitt2008moral} that we include for completeness.

\begin{lemma}[Pointwise minimization]
\label{lem:lagrangian}
For $\lambda\geq0$ and $\mu\in\mathbb R$, the almost everywhere unique
minimizer of the lagrangian $\widehat{\mathcal L}(v,\lambda,\mu)$ over $\mathcal V$ is
$v_{\lambda,\mu}$.
\end{lemma}

\begin{proof}
Up to terms independent of $v$, the Lagrangian is the integral with measure
$f(y\mid a_0)d\nu(y)$ of
\[
    k(v(y))-w_0-[\lambda+\mu s(y)]v(y).
\]
This expression is strictly convex in $v(y)$.  Its interior first-order
condition is $k'(v(y))=\lambda+\mu s(y)$.  Taking limited liability into account,
the solution is $v(y)=\ell(\lambda+\mu s(y))$.
The assumption that canonical contracts are feasible ensures that this solution belongs to $\mathcal V$.  Strict convexity gives almost
everywhere uniqueness.
\end{proof}

\begin{lemma}[The LIC multiplier]
\label{lem:mu-tilde}
For each $\lambda\geq0$, there is a unique $\widetilde\mu(\lambda)>0$ such
that the canonical contract $v_{\lambda,\widetilde\mu(\lambda)}$ satisfies
\eqref{LIC_hat}.  Moreover, $\widetilde\mu$ is continuous.
\end{lemma}

\begin{proof}
For the proof, define
\[
    I(\lambda,\mu)
    :=\int \ell(\lambda+\mu s(y))s(y)f(y\mid a_0)\,d\nu(y).
\]
Because the score has mean zero, this can equivalently be written as
\begin{equation}
 I(\lambda,\mu)
 =\int[\ell(\lambda+\mu s(y))-\ell(\lambda)]s(y)
 f(y\mid a_0)\,d\nu(y).
 \label{eq:I-centered}
\end{equation}
Equation \eqref{LIC_hat} is the same as $I(\lambda,\mu)=c'(a_0)$.

Equation \eqref{eq:I-centered} shows that $I(\lambda,0)=0$ and that
$I(\lambda,\mu)$ is weakly increasing in $\mu$ because the term in brackets has the same sign as the score.  As $\mu\to\infty$,
monotone convergence yields
\[
 I(\lambda,\mu)\longrightarrow
 \Delta u\int_{\{s>0\}}s(y)f(y\mid a_0)\,d\nu(y).
\]
This limit exceeds $c'(a_0)$ by
\eqref{eq:local-implementability}.  Continuity therefore gives a positive
solution.

Whenever $I(\lambda,\mu)>0$, it is strictly increasing in $\mu$.
Therefore the multiplier satisfying \eqref{LIC_hat} is unique.  Finally, joint continuity
of $I$ and this strict crossing imply continuity of
$\widetilde\mu(\lambda)$.
\end{proof}

For each $\lambda\geq0$, define the canonical contract obtained by setting
$\mu=\widetilde\mu(\lambda)$ and the attained utility and cost by:
\begin{eqnarray*}
    v_\lambda(y)&:=&\ell\bigl(\lambda+\widetilde\mu(\lambda)s(y)\bigr),\\
    \widetilde U(\lambda)&:=&\widehat U(v_\lambda,a_0),\\
    \widetilde C(\lambda)&:=&\widehat C(v_\lambda,a_0).
\end{eqnarray*}

The canonical contracts solving local incentive compatibility therefore form a one-dimensional path indexed by
$\lambda$.  The next lemma shows that these contracts solve a Pareto problem indexed by $\lambda$.

\begin{lemma}[Canonical contracts solve a Pareto problem]
\label{lem:pareto-problem}
For every $\lambda\geq0$, $v_\lambda$ is the almost everywhere unique
minimizer of
\[
    \widehat C(v,a_0)-\lambda\widehat U(v,a_0)
\]
among contracts in $\mathcal V$ satisfying \eqref{LIC_hat}.  The function
$\widetilde U$ is
continuous and nondecreasing.  If $\lambda_1<\lambda_2$ and
$\widetilde U(\lambda_1)=\widetilde U(\lambda_2)$, then
$v_{\lambda_1}=v_{\lambda_2}$ almost everywhere.
\end{lemma}

\begin{proof}
On the constraint \eqref{LIC_hat}, the displayed objective differs by a constant from the
Lagrangian with $\mu=\widetilde\mu(\lambda)$.  Lemma
\ref{lem:lagrangian} therefore proves optimality and uniqueness.

For $\lambda_1<\lambda_2$, optimality of $v_{\lambda_1}$ at
$\lambda_1$ and of $v_{\lambda_2}$ at $\lambda_2$ gives
\begin{align*}
    \widetilde C(\lambda_1)-\lambda_1\widetilde U(\lambda_1)
    &\leq
    \widetilde C(\lambda_2)-\lambda_1\widetilde U(\lambda_2),\\
    \widetilde C(\lambda_2)-\lambda_2\widetilde U(\lambda_2)
    &\leq
    \widetilde C(\lambda_1)-\lambda_2\widetilde U(\lambda_1).
\end{align*}
Adding these inequalities gives
\[
    (\lambda_2-\lambda_1)
    [\widetilde U(\lambda_2)-\widetilde U(\lambda_1)]\geq0.
\]
If the utility levels are equal, both contracts minimize the same strictly
convex Pareto problem and hence agree almost everywhere.  Continuity follows
from Lemma \ref{lem:mu-tilde} and canonical-contract regularity.
\end{proof}

Define
\[
    U_L:=\widetilde U(0).
\]
Recall that $U_R$ is the supremum feasible reservation utility for the relaxed
problem.  The next lemma shows that the Pareto path converges to this frontier
and rules out a terminal flat portion.

\begin{lemma}[The right utility frontier]
\label{lem:no-terminal-flat}
We have
\[
    \lim_{\lambda\to\infty}\widetilde U(\lambda)=U_R.
\]
Moreover, for every finite $M$, $\widetilde U(M)<U_R$.
\end{lemma}

\begin{proof}
Let
\[
    U_\infty:=\lim_{\lambda\to\infty}\widetilde U(\lambda),
\]
which is well defined in $\mathbb R\cup\{\infty\}$ by monotonicity.  Since
$v_\lambda$ is feasible for reservation utility $\widetilde U(\lambda)$, we
have $\widetilde U(\lambda)\leq U_R$ for every $\lambda$, and hence
$U_\infty\leq U_R$.

Conversely, for any contract $v$ satisfying \eqref{LIC_hat}, Lemma
\ref{lem:pareto-problem} gives, for every $\lambda>0$,
\[
    \widehat U(v,a_0)
    \leq \widetilde U(\lambda)
       +\frac{\widehat C(v,a_0)-\widetilde C(\lambda)}{\lambda}
    \leq \widetilde U(\lambda)
       +\frac{\widehat C(v,a_0)}{\lambda}.
\]
Letting $\lambda\to\infty$ gives
$\widehat U(v,a_0)\leq U_\infty$.  Thus no reservation utility above
$U_\infty$ is feasible, so $U_R\leq U_\infty$.  Therefore
$U_\infty=U_R$.

Suppose that $\widetilde U(M)=U_R$ for some finite $M$.  Monotonicity makes
$\widetilde U$ constant on $[M,\infty)$, so Lemma
\ref{lem:pareto-problem} implies that every $v_\lambda$, $\lambda\geq M$,
agrees almost everywhere with a fixed contract $v$.  Local implementability
implies that $\{s(y)>0\}$ has positive measure.  On this set,
\[
    v(y)
    =\ell\bigl(\lambda+\widetilde\mu(\lambda)s(y)\bigr)
    \geq\ell(\lambda)
\]
almost everywhere.  Letting $\lambda\to\infty$ gives
$v(y)\geq\bar u$, a contradiction with $v$ being feasible.
\end{proof}

\begin{proof}[Proof of Proposition \ref{prop:relaxed-optimal-contract}]
\emph{Part (i).}  In utility units, the objective is strictly convex and the
constraints are linear, so the relaxed problem has an almost everywhere unique solution.  We
now construct it.  If $\bar U\leq U_L$, then $v_0$ is feasible, and Lemma
\ref{lem:pareto-problem} with $\lambda=0$ says that it minimizes cost among all
contracts satisfying \eqref{LIC_hat}.  Hence it solves the relaxed problem.

Now suppose $\bar U\in(U_L,U_R)$.  Continuity of $\widetilde U$ gives
$\lambda>0$ such that $\widetilde U(\lambda)=\bar U$.  For any relaxed-feasible
contract $v$, Lemma \ref{lem:pareto-problem} gives
\begin{align*}
    \widehat C(v,a_0)
    &\geq \widetilde C(\lambda)
      +\lambda[\widehat U(v,a_0)-\widetilde U(\lambda)]\\
    &\geq \widetilde C(\lambda).
\end{align*}
Thus $v_\lambda$ solves the relaxed problem.  Equality requires $v$ to minimize
the same strictly convex Pareto problem, so $v=v_\lambda$ almost everywhere.
By construction,
$v_\lambda=v_{\lambda,\widetilde\mu(\lambda)}$ is canonical.  Translating back
through $w=k(v)-w_0$ proves part (i).

\emph{Part (ii).}  Part (i) shows that the relaxed-optimal contract is $v_0$
for every $\bar U\leq U_L$, and that IR binds when
$\bar U\in(U_L,U_R)$.  On the latter region, let
\[
    \Lambda(\bar U)
    :=\{\lambda\geq0:\widetilde U(\lambda)=\bar U\}.
\]
Every $\lambda\in\Lambda(\bar U)$ supports the relaxed solution by the argument
in part (i).  Conversely, Lemmas \ref{lem:lagrangian} and
\ref{lem:mu-tilde} imply that a relaxed solution supported by an IR multiplier
$\lambda$ is $v_\lambda$; binding IR then gives
$\widetilde U(\lambda)=\bar U$.  Thus $\Lambda(\bar U)$ is exactly the set of
associated IR multipliers.  Continuity and monotonicity of $\widetilde U$ make
this set a nonempty closed interval, and it is bounded because $\bar U<U_R$.
Hence it has a smallest element.

If $U_L<T_1<T_2<U_R$, monotonicity of $\widetilde U$ gives
\[
    \lambda_1<\lambda_2
    \qquad
    \text{for every }\lambda_i\in\Lambda(T_i).
\]
Moreover, for every finite $M$, Lemma \ref{lem:no-terminal-flat} gives
$\widetilde U(M)<U_R$.  Hence, once $\bar U>\widetilde U(M)$, every
$\lambda\in\Lambda(\bar U)$ exceeds $M$.  Therefore
\[
    \min\Lambda(\bar U)\longrightarrow\infty
    \quad\text{as }\bar U\uparrow U_R.
\]
This proves part (ii).

\emph{Part (iii).}  The value $C^*$ is constant below $U_L$ by part (ii).  To
show strict convexity on $(U_L,U_R)$, take $T_1\neq T_2$ in this interval and
let $v_1,v_2$ be their optimal contracts.  IR binds, so these contracts are
distinct.  For $t\in(0,1)$, the contract
$v_t:=tv_1+(1-t)v_2$ satisfies LIC and delivers utility
$T_t:=tT_1+(1-t)T_2$.  Strict convexity of $k$ therefore gives
\[
    C^*(T_t)
    \leq \widehat C(v_t,a_0)
    <tC^*(T_1)+(1-t)C^*(T_2).
\]

Finally, suppose $U_L<T_1<T_2<U_R$ and take
$\lambda_1\in\Lambda(T_1)$.  Lemma \ref{lem:pareto-problem} gives
\[
    C^*(T_2)-\lambda_1T_2
    \geq C^*(T_1)-\lambda_1T_1.
\]
Since $T_1>U_L$, we have $\lambda_1>0$, and hence
$C^*(T_2)>C^*(T_1)$.  Thus $C^*$ is strictly increasing and strictly convex on
$(U_L,U_R)$, proving part (iii).
\end{proof}

\subsection{Safe tails and concavity}
\label{subsec:appendix-safe-tail}

This subsection proves Proposition \ref{prop:concave}.  We retain the cutoff
itself in the proof because doing so handles bounded supports, nonmonotone
scores, and score atoms without separate arguments.

For a safe cutoff $q\in\mathcal Q_{\mathrm{safe}}$, define
\begin{align}
    \Capq(q)&:=-\int_{\{s(y)\leq q\}}s(y)f(y\mid a_0)\,d\nu(y),
    \label{eq:cutoff-capacity}\\
    \Curvq(q)&:=\sup_{a\in\mathcal A}
    \int_{\{s(y)>q\}}[s(y)f_{aa}(y\mid a)]_+\,d\nu(y).
    \label{eq:cutoff-curvature}
\end{align}
The weak inequality assigns an atom at $q$ to the safe, no-payment side; the
strict inequality in \eqref{eq:cutoff-curvature} is its complement.  As $q$
increases through safe cutoffs, $\Capq(q)$ increases and $\Curvq(q)$ decreases.
Moreover,
\[
 \sup_{q<\bar q_{\mathrm{safe}},\ q\in\mathcal Q_{\mathrm{safe}}}\Capq(q)
 =\mathrm{Cap}_{\mathrm{safe}},
 \qquad
 \inf_{q\in\mathcal Q_{\mathrm{safe}}}\Curvq(q)
 =\mathrm{Curv}_{\mathrm{safe}}.
\]
The strict set in the definition of $\mathrm{Cap}_{\mathrm{safe}}$ is what
makes the first identity valid even at a rightmost safe score atom.

For $\lambda>m_0$ and $q<0$, define the trial multiplier
\begin{equation}
    \mu_q(\lambda):=\frac{m_0-\lambda}{q}>0.
    \label{eq:trial-multiplier}
\end{equation}
It places the limited-liability kink exactly at $q$.

\begin{lemma}[Kink placement]
\label{lem:kink-placement}
Fix $q<0$.  If
\[
    \Delta u\,\Capq(q)>c'(a_0),
\]
where an infinite utility spread times positive capacity is interpreted as
infinity, then, for all sufficiently large $\lambda$,
\[
    \widetilde\mu(\lambda)\leq\mu_q(\lambda)=O(\lambda),
    \qquad
    \frac{m_0-\lambda}{\widetilde\mu(\lambda)}\leq q.
\]
Tail safety is not needed for this lemma.
\end{lemma}

\begin{proof}
At $\mu_q(\lambda)$, scores weakly below $q$ receive $\underline u$.
Centering the local-incentive expression at $\ell(\lambda)$ gives
\[
 \int[\ell(\lambda+\mu_q(\lambda)s(y))-\ell(\lambda)]s(y)
 f(y\mid a_0)\,d\nu(y)
 \geq[\ell(\lambda)-\underline u]\Capq(q).
\]
The omitted contributions are nonnegative: on negative scores both the utility
difference and the score are nonpositive, while on positive scores both are
nonnegative.  As $\lambda\to\infty$, the right-hand side converges to
$\Delta u\,\Capq(q)$ and therefore eventually exceeds $c'(a_0)$.  Monotonicity
of the local-incentive expression in $\mu$ and uniqueness of the LIC multiplier
imply $\widetilde\mu\leq\mu_q$.  Since $m_0-\lambda<0$, division by the smaller
positive multiplier moves the cutoff weakly to the left, proving the second
inequality.
\end{proof}

Fix $\lambda>m_0$, abbreviate
$\widetilde\mu=\widetilde\mu(\lambda)$, and set
\[
    D_\lambda\ell(y):=
    \frac{\ell(\lambda+\widetilde\mu s(y))-\ell(\lambda)}
         {\widetilde\mu s(y)},
\]
with the continuous extension at $s(y)=0$.

\begin{lemma}[Curvature identity]
\label{lem:curvature-identity}
For $\lambda>m_0$ and every $a\in\mathcal A$,
\begin{equation}
 U_{aa}(w_{\lambda,\widetilde\mu},a)
 =\widetilde\mu\int D_\lambda\ell(y)s(y)f_{aa}(y\mid a)\,d\nu(y)-c''(a).
 \label{eq:curvature-identity}
\end{equation}
\end{lemma}

\begin{proof}
Add and subtract the constant $\ell(\lambda)$ and use
$\int f_{aa}(y\mid a)d\nu(y)=0$.  The maintained differentiation condition
then gives the identity.
\end{proof}

\begin{lemma}[Curvature split]
\label{lem:curvature-split}
Fix $\lambda>m_0$ and suppose \eqref{condition:U1} holds.  If
$q\in\mathcal Q_{\mathrm{safe}}$, $\Curvq(q)<\infty$, and the
limited-liability cutoff is weakly below $q$, then, for every
$a\in\mathcal A$,
\begin{equation}
 U_{aa}(w_{\lambda,\widetilde\mu},a)
 \leq
 \widetilde\mu\,\ell_0'(\lambda+\widetilde\mu q)\Curvq(q)-c''(a).
 \label{eq:curvature-split}
\end{equation}
At $m_0$, the derivative is the right derivative.
\end{lemma}

\begin{proof}
Because $A>0$, condition \eqref{condition:U1} entails $P>0$ and is equivalent
to $3A\geq P$.  On the nonbinding region,
\[
    \frac{d}{dW}\log\ell_0'(z(W))=P(W)-3A(W),
    \qquad z(W)=\frac1{u'(W)}.
\]
Since $z$ is increasing in $W$, this proves that $\ell_0$ is concave; it is
increasing because utility is strictly increasing.

On $\{s\leq q\}$, tail safety gives $f_{aa}\geq0$, while $s\leq0$ and
$D_\lambda\ell\geq0$.  This region therefore contributes weakly negatively
to \eqref{eq:curvature-identity}.  On $\{s>q\}$ the contract is above the
limited-liability floor.  Concavity of $\ell_0$ bounds every relevant secant
slope by $\ell_0'(\lambda+\widetilde\mu q)$.  Keeping only the positive part
of $s f_{aa}$ and using \eqref{eq:cutoff-curvature} proves
\eqref{eq:curvature-split}.
\end{proof}

The two preceding lemmas imply the following finite-$\lambda$ bound, which is
the precise version of the argument in the main text.

\begin{proposition}[Finite-$\lambda$ safe-tail bound]
\label{prop:finite-lambda-bound}
Suppose \eqref{condition:U1} holds.  Let
$q_0<q_1\leq0$ be safe cutoffs such that
\[
    \Delta u\,\Capq(q_0)>c'(a_0),
    \qquad \Curvq(q_1)<\infty.
\]
For all sufficiently large $\lambda$,
\begin{equation}
 \sup_{a\in\mathcal A}U_{aa}(w_{\lambda,\widetilde\mu},a)
 \leq
 \mu_{q_0}(\lambda)
 \ell_0'\bigl(\lambda+\mu_{q_0}(\lambda)q_1\bigr)\Curvq(q_1)
 -\underline c''.
 \label{eq:finite-lambda-bound}
\end{equation}
\end{proposition}

\begin{proof}
Lemma \ref{lem:kink-placement} puts the kink weakly below $q_0$ and gives
$\widetilde\mu\leq\mu_{q_0}$.  Apply Lemma
\ref{lem:curvature-split} at $q_1$.  Because $q_1\leq0$ and $\ell_0'$ is
nonincreasing, the map
\[
    \mu\longmapsto \mu\ell_0'(\lambda+\mu q_1)
\]
is nondecreasing on the relevant nonbinding range.  Replacing
$\widetilde\mu$ by $\mu_{q_0}$ and using
$c''(a)\geq\underline c''$ gives \eqref{eq:finite-lambda-bound}.
\end{proof}

We now verify that the right-hand side is negative under each branch of the
main theorem.  On the nonbinding region, if $W=L_0(z)$, differentiation gives
\begin{equation}
    L_0'(z)=\frac{u'(W)}{A(W)},
    \qquad z\ell_0'(z)=L_0'(z).
    \label{eq:link-derivatives}
\end{equation}
Thus \eqref{condition:U2} is equivalent to
$z\ell_0'(z)\to0$.

\begin{lemma}[Sufficient-risk-aversion branch]
\label{lem:risk-aversion-branch}
Under branch (i) of Theorem \ref{thm:main},
\[
    \sup_{a\in\mathcal A}U_{aa}(w_{\lambda,\widetilde\mu},a)<0
\]
for all sufficiently large $\lambda$.
\end{lemma}

\begin{proof}
The strict aggregate capacity inequality and the endpoint convention provide a
safe $q_0<0$ with
$\Delta u\,\Capq(q_0)>c'(a_0)$.  Since safe cutoffs form a lower interval and
$\mathrm{Curv}_{\mathrm{safe}}<\infty$, we may choose a safe
$q_1\in(q_0,0]$ with $\Curvq(q_1)<\infty$.  Put
$z_{01}(\lambda)=\lambda+\mu_{q_0}(\lambda)q_1$.  Then
\[
    \mu_{q_0}(\lambda)=O(\lambda),
    \qquad
    z_{01}(\lambda)\sim
    \lambda\left(1-\frac{q_1}{q_0}\right),
\]
and the coefficient in parentheses is strictly positive.  By
\eqref{eq:link-derivatives},
\[
 \mu_{q_0}(\lambda)\ell_0'(z_{01}(\lambda))\Curvq(q_1)
 \longrightarrow0.
\]
Proposition \ref{prop:finite-lambda-bound} and $\underline c''>0$ complete the
proof.
\end{proof}

\begin{lemma}[Log-utility branch]
\label{lem:log-branch}
Under branch (ii) of Theorem \ref{thm:main},
\[
    \sup_{a\in\mathcal A}U_{aa}(w_{\lambda,\widetilde\mu},a)<0
\]
for all sufficiently large $\lambda$.
\end{lemma}

\begin{proof}
For log utility, $\ell_0'(z)=1/z$ and $\Delta u=\infty$.  The strict
curvature--width inequality, together with the definitions as an infimum and a
supremum, permits safe cutoffs $q_0<q_1\leq0$ such that
\[
    \Capq(q_0)>0,
    \qquad
    \frac{\Curvq(q_1)}{q_1-q_0}<\underline c''.
\]
This remains true with score atoms: choose $q_0$ inside the essential support,
and assign a cutoff atom to the weakly safe side as in
\eqref{eq:cutoff-capacity}.  Since log utility is unbounded above, the first
inequality is precisely the capacity condition in Lemma
\ref{lem:kink-placement}.  Moreover,
\[
 \mu_{q_0}(\lambda)
 \ell_0'(\lambda+\mu_{q_0}(\lambda)q_1)
 =\frac{\mu_{q_0}(\lambda)}
 {\lambda+\mu_{q_0}(\lambda)q_1}
 \longrightarrow\frac1{q_1-q_0}.
\]
The limiting upper bound in Proposition \ref{prop:finite-lambda-bound} is
strictly negative.
\end{proof}

\begin{proof}[Proof of Proposition \ref{prop:concave}]
Lemma \ref{lem:mu-tilde} shows that every canonical contract satisfying
\eqref{LIC_hat} at
a given $\lambda\geq0$ uses the unique multiplier
$\widetilde\mu(\lambda)$.  Lemma \ref{lem:risk-aversion-branch} or Lemma
\ref{lem:log-branch} therefore supplies a finite $\lambda_0$ such that
\[
    U_{aa}(w_{\lambda,\widetilde\mu(\lambda)},a)\leq0
    \quad\text{for every }a\in\mathcal A
\]
whenever $\lambda\geq\lambda_0$.  This is the claim.
\end{proof}

\subsection{An asymptotic-affinity extension}
\label{subsec:asymptotic-affinity}

The safe-tail proof is not the only route to eventual concavity.  This
appendix-only extension covers CRRA coefficients between $1/2$ and $1$ under a
moment condition and concavity of the expected score.  It is stated separately
because its mechanism differs from Proposition \ref{prop:concave}.

\begin{assumption}[Affine-score curvature]
\label{ass:affine-score}
For every $a\in\mathcal A$,
\begin{equation}
    \int s(y)f_{aa}(y\mid a)\,d\nu(y)\leq0.
    \label{eq:affine-score}
\end{equation}
Equivalently, $a\mapsto\E_a[s(Y)]$ is concave.
\end{assumption}

\begin{assumption}[Moments for asymptotic affinity]
\label{ass:affinity-moments}
For some $\eta>0$,
\begin{align*}
 0<\sigma_s^2&:=\int s(y)^2f(y\mid a_0)\,d\nu(y)<\infty,\\
 \int |s(y)|^{2+\eta}f(y\mid a_0)\,d\nu(y)&<\infty,\\
 \sup_{a\in\mathcal A}\int
 (1+|s(y)|^{2+\eta})|f_{aa}(y\mid a)|\,d\nu(y)&<\infty.
\end{align*}
\end{assumption}

Suppose $u(W)=W^{1-\gamma}/(1-\gamma)$, $w_0>0$, and
$\gamma\in(1/2,1)$.  Put $p:=(1-\gamma)/\gamma\in(0,1)$.  On the nonbinding
region,
\begin{equation}
    \ell_0(z)=\frac{z^p}{1-\gamma},
    \qquad
    \ell_0'(z)=\frac1\gamma z^{p-1}.
    \label{eq:crra-link-affinity}
\end{equation}

\begin{lemma}[LIC multiplier asymptotics]
\label{lem:affinity-multiplier}
Under Assumption \ref{ass:affinity-moments},
\[
    \widetilde\mu(\lambda)\ell_0'(\lambda)
    \longrightarrow \frac{c'(a_0)}{\sigma_s^2},
    \qquad
    \frac{\widetilde\mu(\lambda)}{\lambda}\longrightarrow0.
\]
\end{lemma}

\begin{proof}
For fixed $\theta>0$, set
$\mu_\lambda(\theta):=\theta/\ell_0'(\lambda)$.  Since
$\lambda\ell_0'(\lambda)\to\infty$,
$\mu_\lambda(\theta)/\lambda\to0$.  Taylor expansion on
$|s|\leq\lambda/(2\mu_\lambda)$ gives
\[
 \ell(\lambda+\mu_\lambda(\theta)s)
 =\ell(\lambda)+\theta s+r_\lambda(s),
 \qquad
 \int |r_\lambda(s)s|f(y\mid a_0)\,d\nu(y)\to0.
\]
To justify the last assertion, the central remainder is bounded by a constant
times
\[
    \mu_\lambda^2|\ell_0''(\lambda/2)|s^2.
\]
Truncation of the complement uses the $(2+\eta)$ moment and the sublinear
growth of $z^p$.  Both bounds are uniform
for $\theta$ in compact sets.  After multiplication by $s$ and integration,
the constant term vanishes because the score has mean zero, so
\[
 \int \ell(\lambda+\mu_\lambda(\theta)s(y))s(y)
 f(y\mid a_0)\,d\nu(y)
 \longrightarrow\theta\sigma_s^2.
\]
Monotonicity of the left-hand side in $\mu$ brackets the unique LIC multiplier
between any two choices of $\theta$ surrounding
$c'(a_0)/\sigma_s^2$.  This proves the first limit.  The
second follows after division by
$\lambda\ell_0'(\lambda)\to\infty$.
\end{proof}

\begin{lemma}[Uniform affine residual]
\label{lem:uniform-affine-residual}
Under Assumption \ref{ass:affinity-moments},
\begin{align}
 \sup_{a\in\mathcal A}\Bigg|\int
 &\left[\ell(\lambda+\widetilde\mu(\lambda)s(y))-\ell(\lambda)
 -\widetilde\mu(\lambda)\ell_0'(\lambda)s(y)\right]
 \nonumber\\[-2pt]
 &\hspace{35mm}\times f_{aa}(y\mid a)\,d\nu(y)\Bigg|
 \longrightarrow0.
 \label{eq:uniform-affine-residual}
\end{align}
\end{lemma}

\begin{proof}
Split the integral at
$|s|=\lambda/(2\widetilde\mu)$.  On the central region, Taylor's theorem and
\eqref{eq:crra-link-affinity} bound the residual by a constant times
\[
    \widetilde\mu^2|\ell_0''(\lambda/2)|s^2.
\]
Lemma \ref{lem:affinity-multiplier} makes the coefficient $O(\lambda^{-p})$.
On the complement, use
$|\ell(z)|\leq C(1+|z|^p)$, the fact that
$\widetilde\mu/\lambda\to0$, and the uniform $(2+\eta)$ weighted moment in
Assumption \ref{ass:affinity-moments}.  Markov's inequality (with threshold
$\lambda/(2\widetilde\mu)$) makes every tail term vanish uniformly in $a$.
Combining the two regions proves \eqref{eq:uniform-affine-residual}.
\end{proof}

\begin{proposition}[Asymptotic-affinity extension]
\label{prop:asymptotic-affinity}
Suppose $\underline c''>0$, Assumptions \ref{ass:affine-score} and
\ref{ass:affinity-moments} hold, and utility is CRRA with
$\gamma\in(1/2,1)$.  Then
\[
    \sup_{a\in\mathcal A}
    U_{aa}(w_{\lambda,\widetilde\mu(\lambda)},a)
    \leq-\underline c''+o(1).
\]
Consequently, the agent's objective is concave for all sufficiently large
$\lambda$, and the first-order approach is valid at all sufficiently high
feasible reservation utilities.
\end{proposition}

\begin{proof}
Use $\int f_{aa}(y\mid a)d\nu(y)=0$ to remove $\ell(\lambda)$.  Lemma
\ref{lem:uniform-affine-residual} gives, uniformly in $a$,
\begin{align*}
 \int \ell(\lambda+\widetilde\mu s(y))f_{aa}(y\mid a)\,d\nu(y)
 ={}&\widetilde\mu\ell_0'(\lambda)
 \int s(y)f_{aa}(y\mid a)\,d\nu(y)+o(1).
\end{align*}
The leading term is nonpositive by Assumption \ref{ass:affine-score}.  After
subtracting $c''(a)\geq\underline c''$, the asserted bound follows.  LIC at the
interior action $a_0$ and concavity imply global incentive compatibility.  The
last conclusion then follows from the divergence of the smallest supporting
multiplier in Proposition \ref{prop:relaxed-optimal-contract}.
\end{proof}

\begin{corollary}[Gaussian CRRA extension]
\label{cor:gaussian-crra-extension}
For a Gaussian location family on a compact action set, the moment conditions
in Assumption \ref{ass:affinity-moments} hold and
\[
    \int s(y)f_{aa}(y\mid a)\,d\nu(y)
    =\frac{d^2}{da^2}\E_a[s(Y)]=0.
\]
Hence Gaussian output with CRRA utility is covered by Theorem
\ref{thm:main} for $\gamma>1$, by its log branch for $\gamma=1$, and by
Proposition \ref{prop:asymptotic-affinity} for $\gamma\in(1/2,1)$, subject to
the assumptions of the corresponding result.  No general claim is made here
for $\gamma\leq1/2$.
\end{corollary}

\begin{proof}
For Gaussian location output, $s(y)=(y-a_0)/\sigma^2$ is affine in $y$ and all
of its moments against $f(y\mid a_0)$ and the Gaussian derivatives
$f_{aa}(y\mid a)$ are finite uniformly on compact action sets.  Moreover,
$\E_a[s(Y)]=(a-a_0)/\sigma^2$ is affine in $a$, proving the displayed
identity.  The stated cases now follow from the corresponding results.
\end{proof}
