\section{Derivations of Scores for Particular Distributions}

For each distribution, the action score is
\[
    S(y\mid a):=\partial_a\log f(y\mid a)
    =\frac{f_a(y\mid a)}{f(y\mid a)}.
\]
Evaluating this expression at the intended action gives the score used in the
body of the paper, $s(y)=S(y\mid a_0)$.

\subsection{Gaussian}

For Gaussian output with mean $a$ and variance $\sigma^2$,
\[
    f(y\mid a)
    =\frac{1}{\sqrt{2\pi\sigma^2}}
      \exp\left(-\frac{(y-a)^2}{2\sigma^2}\right).
\]
Therefore
\[
    S(y\mid a)
    =\partial_a\log f(y\mid a)
    =\frac{y-a}{\sigma^2}.
\]

\subsection{Lognormal}

Suppose $\log y$ is Gaussian with mean $a$ and variance $\sigma^2$.  For
$y>0$,
\[
    f(y\mid a)
    =\frac{1}{y\sqrt{2\pi\sigma^2}}
      \exp\left(-\frac{(\log y-a)^2}{2\sigma^2}\right).
\]
Differentiating its log gives
\[
    S(y\mid a)=\frac{\log y-a}{\sigma^2}.
\]

\subsection{Poisson}

For $a>0$ and $y\in\{0,1,\ldots\}$,
\[
    f(y\mid a)=\frac{a^y e^{-a}}{y!}.
\]
Thus
\[
    S(y\mid a)
    =\partial_a\bigl[y\log a-a-\log(y!)\bigr]
    =\frac{y}{a}-1
    =\frac{y-a}{a}.
\]

\subsection{Exponential}

For the exponential distribution parametrized by its mean $a>0$,
\[
    f(y\mid a)=\frac{1}{a}\exp\left(-\frac{y}{a}\right),
    \qquad y\geq0.
\]
Hence
\[
    S(y\mid a)
    =\partial_a\left[-\log a-\frac{y}{a}\right]
    =-\frac{1}{a}+\frac{y}{a^2}
    =\frac{y-a}{a^2}.
\]

\subsection{Bernoulli}

For $a\in(0,1)$ and $y\in\{0,1\}$,
\[
    f(y\mid a)=a^y(1-a)^{1-y}.
\]
Therefore
\[
    S(y\mid a)
    =\frac{y}{a}-\frac{1-y}{1-a}
    =\frac{y-a}{a-a^2}.
\]

\subsection{Geometric}

Consider the geometric distribution on $\{1,2,\ldots\}$ parametrized by its
mean $a>1$:
\[
    f(y\mid a)
    =\left(1-\frac{1}{a}\right)^{y-1}\frac{1}{a}.
\]
Its log density is
\[
    \log f(y\mid a)
    =(y-1)\bigl[\log(a-1)-\log a\bigr]-\log a.
\]
Consequently,
\[
    S(y\mid a)
    =(y-1)\left(\frac{1}{a-1}-\frac{1}{a}\right)-\frac{1}{a}
    =\frac{y-a}{a^2-a}.
\]

\subsection{Binomial}

For $a\in(0,1)$, a fixed number of trials $n$, and
$y\in\{0,\ldots,n\}$,
\[
    f(y\mid a)=\binom{n}{y}a^y(1-a)^{n-y}.
\]
It follows that
\[
    S(y\mid a)
    =\frac{y}{a}-\frac{n-y}{1-a}
    =\frac{y-na}{a-a^2}.
\]

\subsection{Gamma}

For a fixed shape parameter $n>0$ and scale parameter $a>0$,
\[
    f(y\mid a)
    =\frac{y^{n-1}e^{-y/a}}{\Gamma(n)a^n},
    \qquad y>0.
\]
Thus
\[
    S(y\mid a)
    =\partial_a\left[-\frac{y}{a}-n\log a\right]
    =\frac{y}{a^2}-\frac{n}{a}
    =\frac{y-na}{a^2}.
\]

\subsection{Student's $t$}

For location parameter $a$, scale parameter $\sigma>0$, and degrees of
freedom $d>0$,
\[
    f(y\mid a)
    =\frac{\Gamma\left(\frac{d+1}{2}\right)}
           {\Gamma\left(\frac{d}{2}\right)\sqrt{\pi d}\,\sigma}
      \left(1+\frac{(y-a)^2}{d\sigma^2}\right)^{-(d+1)/2}.
\]
Differentiating the log density gives
\[
    S(y\mid a)
    =\frac{(d+1)(y-a)}{d\sigma^2+(y-a)^2}.
\]

\subsection{Exponential Family}

Consider
\[
    f(y\mid a)
    =h(y)\exp\bigl(\eta(a)T(y)-\psi(a)\bigr),
\]
where $h$ is the base density or mass function, $\eta$ is the natural
parameter, $T$ is the sufficient statistic, and $A$ is the log-normalizing
term expressed as a function of action.  Then
\[
    S(y\mid a)
    =\partial_a\log f(y\mid a)
    =\eta'(a)T(y)-\psi'(a).
\]

\subsection{Additive Model}

Let $Y=a+X$, where $X$ has differentiable density $h$.  Then
\[
    f(y\mid a)=h(y-a),
    \qquad
    f_a(y\mid a)=-h'(y-a).
\]
Therefore
\[
    S(y\mid a)=-\frac{h'(y-a)}{h(y-a)}.
\]

\subsection{Multiplicative Model}

Let $Y=aX$, where $a>0$ and $X$ has differentiable density $h$.  A change of
variables gives
\[
    f(y\mid a)=\frac{1}{a}h\left(\frac{y}{a}\right).
\]
Differentiating with respect to $a$ yields
\[
    f_a(y\mid a)
    =-\frac{1}{a^2}h\left(\frac{y}{a}\right)
     -\frac{y}{a^3}h'\left(\frac{y}{a}\right).
\]
Hence
\[
    S(y\mid a)
    =-\frac{1}{a}
     -\frac{y}{a^2}\frac{h'(y/a)}{h(y/a)}.
\]

\section{Derivations of Link Functions for Particular Utility Functions}

The body of the paper defines utility over total wealth $W$.  Let
\[
    m_0:=\frac{1}{u'(w_0)}.
\]
On the region where limited liability does not bind, the first-order condition
is $u'(W)=1/z$.  The base wage link function and limited-liability wage link function are
therefore
\[
    L_0(z):=(u')^{-1}(1/z),
    \qquad
    L(z):=L_0(z\vee m_0).
\]
For $z(y)=\lambda+\mu s(y)$, the canonical total wealth and wage are
\[
    W_{\lambda,\mu}(y)=L(z(y)),
    \qquad
    w_{\lambda,\mu}(y)=L(z(y))-w_0.
\]
The proofs appendix also uses the corresponding utility link function
$\ell(z):=u(L(z))$.  We report it below to distinguish it from the wage link
function $L$ used in the body.

\subsection{Log}

Let
\[
    u(W)=\log W,
    \qquad W>0,
\]
with $w_0>0$.  Since $u'(W)=1/W$, the equation $u'(W)=1/z$ gives
$L_0(z)=z$.  Moreover,
\[
    m_0=\frac{1}{u'(w_0)}=w_0.
\]
Thus
\[
    L(z)=z\vee w_0,
    \qquad
    \ell(z)=\log(z\vee w_0).
\]
The associated canonical wage is
\[
    w_{\lambda,\mu}(y)
    =\bigl[\lambda+\mu s(y)-w_0\bigr]^+.
\]

\subsection{CRRA}

Let
\[
    u(W)=\frac{W^{1-\gamma}}{1-\gamma},
    \qquad W>0,
\]
where $\gamma>0$ and $\gamma\neq1$.  Since $u'(W)=W^{-\gamma}$, the equation
$u'(W)=1/z$ gives
\[
    L_0(z)=z^{1/\gamma}.
\]
Also,
\[
    m_0=\frac{1}{u'(w_0)}=w_0^\gamma.
\]
Consequently,
\[
    L(z)=\bigl(z\vee w_0^\gamma\bigr)^{1/\gamma},
    \qquad
    \ell(z)
    =\frac{\bigl(z\vee w_0^\gamma\bigr)^{(1-\gamma)/\gamma}}
           {1-\gamma}.
\]
The associated canonical wage is
\[
    w_{\lambda,\mu}(y)
    =\Bigl(\bigl[\lambda+\mu s(y)\bigr]\vee w_0^\gamma\Bigr)^{1/\gamma}-w_0.
\]

\subsection{CARA}

Let
\[
    u(W)=-\frac{\exp(-\alpha W)}{\alpha},
\]
where $\alpha>0$.  Since $u'(W)=\exp(-\alpha W)$, the equation
$u'(W)=1/z$ gives
\[
    L_0(z)=\frac{\log z}{\alpha}.
\]
Furthermore,
\[
    m_0=\frac{1}{u'(w_0)}=\exp(\alpha w_0).
\]
It follows that
\[
    L(z)=\frac{\log\bigl(z\vee\exp(\alpha w_0)\bigr)}{\alpha},
    \qquad
    \ell(z)=-\frac{1}{\alpha\bigl(z\vee\exp(\alpha w_0)\bigr)}.
\]
The associated canonical wage is
\[
    w_{\lambda,\mu}(y)
    =\frac{\log\Bigl(\bigl[\lambda+\mu s(y)\bigr]
                       \vee\exp(\alpha w_0)\Bigr)}{\alpha}
     -w_0.
\]

\section{Verification of the Common-Distribution and Common-Utility Remarks}
\label{sec:common-regularity-verification}

This section verifies the two remarks following the maintained assumptions in
the body of the paper.  These verifications supply the distributional and
utility calculations used in Corollary~\ref{cor:common-utilities}.

\subsection{Safe tails in common distributions}

We first verify Remark~\ref{rem:safe-tails-common-distributions}.  Recall the
identity
\begin{equation}
    \frac{f_{aa}(y\mid a)}{f(y\mid a)}
    =S(y\mid a)^2+\partial_a S(y\mid a).
    \label{eq:faa-score-identity}
\end{equation}
Because the action set is compact and lies in the interior of the parameter
space under the restrictions in the remark, all parameter-dependent
coefficients below are uniformly bounded.

For Gaussian output,
\[
 \frac{f_{aa}(y\mid a)}{f(y\mid a)}
 =\frac{(y-a)^2}{\sigma^4}-\frac1{\sigma^2}.
\]
Consequently, a sufficiently far-left half-line is safe uniformly in
$a\in\mathcal A$.  Since the Gaussian score is affine in $y$ and is
unbounded below, the safe width is infinite.

For the continuous nonnegative-output families, equation
\eqref{eq:faa-score-identity} gives, at the lower endpoint $y=0$,
\[
 \left.\frac{f_{aa}}f\right|_{y=0}
 =\frac{2}{a^2}\quad\text{for exponential output},
 \qquad
 \left.\frac{f_{aa}}f\right|_{y=0}
 =\frac{n(n+1)}{a^2}\quad\text{for gamma output}.
\]
These expressions are uniformly positive.  Continuity and compactness
therefore give an $\varepsilon>0$ such that $f_{aa}(y\mid a)\geq0$ for every
$a\in\mathcal A$ and $0<y\leq\varepsilon$.  Because the score is increasing
and strictly negative near zero, this interval contains a safe low-score
region of positive width and positive probability.

For the discrete families, it is enough to consider the lowest outcome.  Its
probability and second action derivative are
\begin{center}
\begin{tabular}{lcc}
\toprule
Family & lowest-outcome probability & second derivative \\
\midrule
Poisson & $f(0\mid a)=e^{-a}$ & $f_{aa}(0\mid a)=e^{-a}$ \\
Geometric & $f(1\mid a)=a^{-1}$ & $f_{aa}(1\mid a)=2a^{-3}$ \\
Bernoulli & $f(0\mid a)=1-a$ & $f_{aa}(0\mid a)=0$ \\
Binomial & $f(0\mid a)=(1-a)^n$ &
$n(n-1)(1-a)^{n-2}$ \\
\bottomrule
\end{tabular}
\end{center}
Thus the lowest outcome is safe and has a strictly negative score and positive
probability.  Choose a negative cutoff strictly between its score and the
score of the next outcome.  The resulting safe region consists only of the
lowest outcome, proving positive safe capacity and positive safe width.

It remains to verify finite safe curvature.  In the Gaussian, Poisson,
exponential, and gamma cases, $f_{aa}/f$ is a polynomial of degree at most two
in the output, while the score is affine in the output.  Compactness of the
parameter set and the standard uniform moment bounds therefore imply
\[
 \sup_{a\in\mathcal A}\int |s(y)f_{aa}(y\mid a)|\,d\nu(y)<\infty.
\]
The same conclusion is immediate for Bernoulli and binomial output.  For
geometric output, $f_{aa}/f$ is also at most quadratic in the outcome and the
geometric moments are uniformly bounded when the compact action set is
contained in $(1,\infty)$.  This bound implies
$\mathrm{Curv}_{\mathrm{safe}}<\infty$ in every case.  The safe regions just
constructed have positive probability and contain only negative scores, so
they also imply $\mathrm{Cap}_{\mathrm{safe}}>0$.  This proves all the claims
for the named distributions.

For completeness, consider the regular exponential family in the remark:
\[
 f(y\mid a)=h(y)\exp\bigl(\eta(a)T(y)-\psi(a)\bigr).
\]
Direct differentiation gives
\begin{equation}
 \frac{f_{aa}}f
 =\bigl(\eta'(a)T-\psi'(a)\bigr)^2+\eta''(a)T-\psi''(a).
 \label{eq:exponential-family-faa}
\end{equation}
The positive quadratic term dominates uniformly as $T\to-\infty$ because
$\inf_a\eta'(a)>0$.  Hence a sufficiently low-score tail is safe.  Essential
unboundedness of $T$ below makes its safe width infinite and gives that tail
positive probability, hence positive capacity.  Finally, because the compact
set $\eta(\mathcal A)$ lies in the interior of the natural-parameter space,
the required polynomial moments of $T$ are uniformly finite.  Equation
\eqref{eq:exponential-family-faa} then gives finite safe curvature.

\subsection{Sufficient risk aversion for common utilities}

We next verify the utility remark following Condition
\ref{condition:sufficient-risk-aversion}.  For CARA utility,
\[
 A(W)=\alpha,\qquad P(W)=\alpha,
 \qquad \frac{A(W)}{u'(W)}=\alpha e^{\alpha W}\longrightarrow\infty.
\]
Thus $A/P=1\geq1/3$ and both parts of the sufficient-risk-aversion condition
hold.  For CRRA utility with coefficient $\gamma$,
\[
 A(W)=\frac{\gamma}{W},\qquad
 P(W)=\frac{\gamma+1}{W},\qquad
 \frac{A(W)}{P(W)}=\frac{\gamma}{\gamma+1},
 \qquad
 \frac{A(W)}{u'(W)}=\gamma W^{\gamma-1}.
\]
When $\gamma>1$, the first ratio exceeds $1/2$ and the final expression tends
to infinity.  Hence CARA and CRRA with $\gamma>1$ satisfy Condition
\ref{condition:sufficient-risk-aversion}, as claimed.

\section{First-Order Approach with CRRA and
\texorpdfstring{$\gamma\in(0,1)$}{gamma in (0, 1)}}
\label{sec:asymptotic-affinity}

We extend the main theorem to CRRA utility with $\gamma\in(0,1)$ under
additional assumptions. We use the payment- and utility-contract notation
introduced in Appendix~\ref{subsec:appendix-notation}. The main additional
restriction is that the expected score is concave in the agent's action. The
argument is that, at high reservation utility, the utility contract is
asymptotically affine in the score. The leading curvature term is then
proportional to the curvature of the expected score and is dominated by the
strict convexity of effort cost.

\begin{assumption}[CRRA utility]
\label{ass:crra-affinity}
Utility is
\[
    u(W)=\frac{W^{1-\gamma}}{1-\gamma},
    \qquad w_0>0,
    \qquad \gamma\in(0,1).
\]
\end{assumption}

Put $p:=(1-\gamma)/\gamma>0$.  On the nonbinding region,
\begin{equation}
    \ell_0(z)=\frac{z^p}{1-\gamma},
    \qquad
    \ell_0'(z)=\frac1\gamma z^{p-1}.
    \label{eq:crra-link-affinity}
\end{equation}

\begin{assumption}[Affine-score curvature]
\label{ass:affine-score}
For every $a\in\mathcal A$,
\begin{equation}
    \int s(y)f_{aa}(y\mid a)\,d\nu(y)\leq0.
    \label{eq:affine-score}
\end{equation}
Equivalently, $a\mapsto\E_a[s(Y)]$ is concave.
\end{assumption}

\begin{assumption}[Moments for asymptotic affinity]
\label{ass:affinity-moments}
For some $r>\max\{2,p+1\}$,
\begin{align*}
 0<\sigma_s^2:=\int s(y)^2f(y\mid a_0)\,d\nu(y)&<\infty,\\
 \int |s(y)|^r f(y\mid a_0)\,d\nu(y)&<\infty,\\
 \sup_{a\in\mathcal A}\int
 (1+|s(y)|^r)|f_{aa}(y\mid a)|\,d\nu(y)&<\infty.
\end{align*}
\end{assumption}

\begin{lemma}[Local incentive compatibility multiplier asymptotics]
\label{lem:affinity-multiplier}
Under Assumptions \ref{ass:crra-affinity} and
\ref{ass:affinity-moments},
\[
    \widetilde\mu(\lambda)\ell_0'(\lambda)
    \longrightarrow \frac{c'(a_0)}{\sigma_s^2},
    \qquad
    \frac{\widetilde\mu(\lambda)}{\lambda}\longrightarrow0.
\]
\end{lemma}

\begin{proof}
For fixed $\theta>0$, set
$\mu_\lambda(\theta):=\theta/\ell_0'(\lambda)$.  Since
$\lambda\ell_0'(\lambda)\to\infty$,
$\mu_\lambda(\theta)/\lambda\to0$.  Taylor expansion on
$|s|\leq\lambda/(2\mu_\lambda)$ gives
\[
 \ell(\lambda+\mu_\lambda(\theta)s)
 =\ell(\lambda)+\theta s+r_\lambda(s),
 \qquad
 \int |r_\lambda(s)s|f(y\mid a_0)\,d\nu(y)\to0.
\]
To justify the last assertion, put
$T_\lambda:=\lambda/(2\mu_\lambda)$.  Uniformly for $\theta$ in compact
subsets of $(0,\infty)$, $T_\lambda\asymp\lambda^p$.  On
$\{|s|\leq T_\lambda\}$, Taylor's theorem bounds $|r_\lambda(s)s|$, for
a constant $K$ independent of $\lambda$ and uniform over $\theta$ in the
chosen compact set, by
\[
    K\mu_\lambda^2|\ell_0''(\lambda/2)||s|^3
    \leq K\lambda^{-p}|s|^3.
\]
The $r$ moment makes the integral of this bound vanish.  If $r\geq3$, this is
$O(\lambda^{-p})$.  If $r<3$, use
$|s|^3\leq T_\lambda^{3-r}|s|^r$ on the central region to obtain
$O(\lambda^{-p(r-2)})$.

On $\{|s|>T_\lambda\}$, polynomial growth gives
\[
 |r_\lambda(s)s|
 \leq K\left[(1+\lambda^p)|s|
       +\mu_\lambda^p|s|^{p+1}+s^2\right].
\]
The $r$ moment, $r>\max\{2,p+1\}$, and
$T_\lambda\asymp\lambda^p$ imply that the integrals of all three terms vanish.
Indeed,
\[
 (1+\lambda^p)T_\lambda^{-(r-1)}
 +\mu_\lambda^pT_\lambda^{-(r-p-1)}
 +T_\lambda^{-(r-2)}
 =O\bigl(\lambda^{-p(r-2)}\bigr),
\]
using $\mu_\lambda^p=O(\lambda^{p(1-p)})$.  These estimates prove the
asserted convergence, uniformly for $\theta$ in compact sets.  After
multiplication by $s$ and integration, the constant term vanishes because the
score has mean zero, so
\[
 \int \ell(\lambda+\mu_\lambda(\theta)s(y))s(y)
 f(y\mid a_0)\,d\nu(y)
 \longrightarrow\theta\sigma_s^2.
\]
Monotonicity of the left-hand side in $\mu$ brackets the unique local
incentive compatibility multiplier
between any two choices of $\theta$ surrounding
$c'(a_0)/\sigma_s^2$.  This proves the first limit.  The
second follows after division by
$\lambda\ell_0'(\lambda)\to\infty$.
\end{proof}

\begin{lemma}[Uniform affine residual]
\label{lem:uniform-affine-residual}
Under Assumptions \ref{ass:crra-affinity} and
\ref{ass:affinity-moments},
\begin{align}
 \sup_{a\in\mathcal A}\Bigg|\int
 &\left[\ell(\lambda+\widetilde\mu(\lambda)s(y))-\ell(\lambda)
 -\widetilde\mu(\lambda)\ell_0'(\lambda)s(y)\right]
 \nonumber\\[-2pt]
 &\hspace{35mm}\times f_{aa}(y\mid a)\,d\nu(y)\Bigg|
 \longrightarrow0.
 \label{eq:uniform-affine-residual}
\end{align}
\end{lemma}

\begin{proof}
Abbreviate $\widetilde\mu=\widetilde\mu(\lambda)$ and set
$T_\lambda:=\lambda/(2\widetilde\mu)$.  Lemma
\ref{lem:affinity-multiplier} gives
$T_\lambda\asymp\lambda^p$ and
$\widetilde\mu\ell_0'(\lambda)=O(1)$.  On $\{|s|\leq T_\lambda\}$,
Taylor's theorem and \eqref{eq:crra-link-affinity} bound the absolute residual,
for a constant $K$ independent of $\lambda$ and $a$, by
\[
    K\widetilde\mu^2|\ell_0''(\lambda/2)|s^2
    \leq K\lambda^{-p}s^2.
\]
Its integral against $|f_{aa}(y\mid a)|$ is therefore $O(\lambda^{-p})$
uniformly in $a$.

On $\{|s|>T_\lambda\}$, the bound
\[
    |\ell(\lambda+\widetilde\mu s)|
    \leq K(1+\lambda^p+\widetilde\mu^p|s|^p)
\]
and $\widetilde\mu\ell_0'(\lambda)=O(1)$ bound the absolute residual by
\[
    K\left[1+\lambda^p+\widetilde\mu^p|s|^p+|s|\right].
\]
Using the uniform $r$ weighted moment in Assumption
\ref{ass:affinity-moments}, its tail integral is bounded by a constant times
\[
 (1+\lambda^p)T_\lambda^{-r}
 +\widetilde\mu^pT_\lambda^{-(r-p)}
 +T_\lambda^{-(r-1)}
 =O\bigl(\lambda^{-p(r-1)}\bigr),
\]
uniformly in $a$.  Combining the central and tail bounds proves
\eqref{eq:uniform-affine-residual}.
\end{proof}

\begin{proposition}[Asymptotic-affinity extension]
\label{prop:asymptotic-affinity}
Suppose $\underline c''>0$ and Assumptions \ref{ass:crra-affinity},
\ref{ass:affine-score}, and \ref{ass:affinity-moments} hold.  Then
\[
    \sup_{a\in\mathcal A}
    \widehat U_{aa}(v_\lambda,a)
    \leq-\underline c''+o(1).
\]
Consequently, the agent's objective is concave for all sufficiently large
$\lambda$, and the first-order approach is valid at all sufficiently high
feasible reservation utilities.
\end{proposition}

\begin{proof}
Abbreviate $\widetilde\mu=\widetilde\mu(\lambda)$.  Using
$\int f_{aa}(y\mid a)d\nu(y)=0$ to remove $\ell(\lambda)$, Lemma
\ref{lem:uniform-affine-residual} gives, uniformly in $a$,
\begin{align*}
 \widehat U_{aa}(v_\lambda,a)
 ={}&\widetilde\mu\ell_0'(\lambda)
 \int s(y)f_{aa}(y\mid a)\,d\nu(y)-c''(a)+o(1).
\end{align*}
The leading term is nonpositive by Assumption \ref{ass:affine-score}, and
$c''(a)\geq\underline c''$, proving the asserted bound.  Local incentive
compatibility at the interior action $a_0$ and concavity imply global
incentive compatibility.  The
last conclusion then follows from the divergence of the smallest supporting
multiplier in Proposition \ref{prop:relaxed-optimal-contract}.
\end{proof}

\begin{corollary}[Common linear-score CRRA extension]
\label{cor:gaussian-crra-extension}
For Gaussian output with fixed positive variance; Poisson and exponential
output with $\mathcal A$ compact and bounded away from $0$; gamma output with
fixed positive shape and $\mathcal A$ compact and bounded away from $0$;
geometric output with $\mathcal A$ compact and contained in $(1,\infty)$; and
Bernoulli and binomial output with $\mathcal A$ compact and contained in
$(0,1)$, the moment conditions in Assumption
\ref{ass:affinity-moments} hold for every $\gamma\in(0,1)$ and
\[
    \int s(y)f_{aa}(y\mid a)\,d\nu(y)
    =\frac{d^2}{da^2}\E_a[s(Y)]=0.
\]
Hence, subject to the maintained assumptions, all the listed output
distributions with CRRA utility are covered by Theorem \ref{thm:main} for
$\gamma>1$ whenever their capacity conditions hold (in particular, for
sufficiently low initial wealth), and by Proposition
\ref{prop:asymptotic-affinity} for $\gamma\in(0,1)$.  At $\gamma=1$, the log
branch of Theorem \ref{thm:main} applies automatically to Gaussian output and,
more generally, to the essentially-unbounded-below regular exponential
families in Remark \ref{rem:safe-tails-common-distributions}.  It does not
automatically cover the other listed families, which must satisfy the
curvature--width inequality \eqref{eq:log-curvature-width-condition}.
\end{corollary}

\begin{proof}
In each case the score at $a_0$ is affine in a statistic whose expectation is
affine in $a$.  For Gaussian, Poisson, exponential, Bernoulli, binomial,
geometric, and gamma output, respectively,
\[
    \E_a[s(Y)]
    =
    \frac{a-a_0}{\sigma^2},\quad
    \frac{a-a_0}{a_0},\quad
    \frac{a-a_0}{a_0^2},\quad
    \frac{a-a_0}{a_0-a_0^2},\quad
    \frac{n(a-a_0)}{a_0-a_0^2},\quad
    \frac{a-a_0}{a_0^2-a_0},\quad
    \frac{n(a-a_0)}{a_0^2}.
\]
This proves the displayed identity.  The required moments are finite uniformly
on the stated compact action sets: trivially for finite support, and by the
standard polynomial moment bounds for Gaussian, Poisson, exponential,
geometric, and gamma families, together with the same bounds applied to the
polynomial multiples of densities that make up $f_{aa}$.  The stated
cases now follow from the corresponding results.
\end{proof}

\section{Proof of Corollary~\ref{cor:common-utilities}}
\label{sec:proof-common-utilities}

\begin{proof}
For part (i), the preceding utility verification shows that CARA and CRRA with
$\gamma>1$ satisfy Condition
\ref{condition:sufficient-risk-aversion}.  Their available utility ranges
above limited liability are, respectively,
\[
 \Delta u=\frac{e^{-\alpha w_0}}{\alpha},
 \qquad
 \Delta u=\frac{w_0^{1-\gamma}}{\gamma-1}.
\]
The first tends to infinity as $w_0\to-\infty$, and the second tends to
infinity as $w_0\downarrow0$.  If
$\mathrm{Cap}_{\mathrm{safe}}>0$, sufficiently low initial wealth therefore
ensures
\[
 \Delta u\,\mathrm{Cap}_{\mathrm{safe}}>c'(a_0).
\]
Theorem~\ref{thm:main}(i) proves the claim.  The named distributions satisfy
the required capacity and curvature properties by the verification above.

For part (ii), essential unboundedness of the score below and the existence of
a safe cutoff imply
$\mathrm{Width}_{\mathrm{safe}}=\infty$.  Consequently,
\[
 \frac{\mathrm{Curv}_{\mathrm{safe}}}
      {\mathrm{Width}_{\mathrm{safe}}}=0<\underline c'',
\]
where the equality uses the finiteness of safe curvature.  The log branch,
Theorem~\ref{thm:main}(ii), proves the claim.  The Gaussian and regular
exponential-family cases satisfy these distributional hypotheses by the
verification above.

Part (iii) follows from Proposition~\ref{prop:asymptotic-affinity} and
Corollary~\ref{cor:gaussian-crra-extension}, whose proof verifies both the
affine-score curvature and uniform moment requirements for every named
family.
\end{proof}

\section{Risk-Neutral Counterexample to Theorem~\ref{thm:main}}
\label{sec:risk-neutral-counterexample}

This section gives a simple example showing that risk aversion cannot be
removed from the main result.  The example satisfies the distributional
conditions used by Theorem~\ref{thm:main}, but risk-neutral utility violates
the maintained strict-concavity and sufficient-risk-aversion assumptions.

Let the agent be risk neutral, $u(W)=W$, with initial wealth $w_0=0$.  Let
output have low, middle, and high states
$\mathcal Y=\{y_-,y_0,y_+\}$, let $\mathcal A=[1/4,3/4]$, and let the
intended action be $a_0=1/2$.  Define
\[
    f(y_+\mid a)=a^3,\qquad
    f(y_-\mid a)=\frac{9}{10}-a,\qquad
    f(y_0\mid a)=\frac{1}{10}+a-a^3.
\]
These probabilities are strictly positive on $\mathcal A$ and sum to one.
Let the effort cost be
\[
    c(a)=5a^2.
\]
Then $c'(a_0)=5$ and $\underline c''=10$.

At $a_0=1/2$, the probabilities and scores are
\begin{align*}
    f(y_+\mid a_0)&=\frac18, & s(y_+)&=6,\\
    f(y_0\mid a_0)&=\frac{19}{40}, & s(y_0)&=\frac{10}{19},\\
    f(y_-\mid a_0)&=\frac25, & s(y_-)&=-\frac52.
\end{align*}
The only low-score state is $y_-$, and $f_{aa}(y_-\mid a)=0$ for every
$a\in\mathcal A$.  Thus every cutoff
$q\in[-5/2,10/19)$ generates the safe low-score region $\{y_-\}$, and
\[
    \mathrm{Cap}_{\mathrm{safe}}
    =-s(y_-)f(y_-\mid a_0)=1>0,
    \qquad
    \mathrm{Width}_{\mathrm{safe}}
    =\frac{10}{19}+\frac52=\frac{115}{38}>0.
\]
The safe curvature is finite.  Since
$f_{aa}(y_+\mid a)=6a$, $f_{aa}(y_0\mid a)=-6a$, and
$f_{aa}(y_-\mid a)=0$,
\[
    \mathrm{Curv}_{\mathrm{safe}}
    =
    \sup_{a\in[1/4,3/4]} s(y_+)f_{aa}(y_+\mid a)
    =
    \sup_{a\in[1/4,3/4]}36a
    =
    27.
\]
In particular,
\[
    \frac{\mathrm{Curv}_{\mathrm{safe}}}
         {\mathrm{Width}_{\mathrm{safe}}}
    =
    \frac{1026}{115}<10=\underline c''.
\]
Thus, if utility were log with positive initial wealth, the log branch of
Theorem~\ref{thm:main} would apply.  The failure below is due to risk
neutrality.

For any reservation utility $\bar U\geq -5/12$, consider the contract
\[
    w(y_-)=w(y_0)=t,\qquad w(y_+)=t+\frac{20}{3},
    \qquad
    t=\bar U+\frac{5}{12}.
\]
The local incentive constraint at $a_0$ is
\[
    \sum_y w(y)f_a(y\mid a_0)=c'(a_0).
\]
The flat transfer $t$ contributes nothing because
$\sum_y f_a(y\mid a_0)=0$, while the bonus on $y_+$ contributes
\[
    \frac{20}{3}f_a(y_+\mid a_0)
    =
    \frac{20}{3}\cdot\frac34
    =
    5=c'(a_0).
\]
Thus the contract satisfies local incentive compatibility.  Individual
rationality binds at $a_0$ because
\[
    t+\frac{20}{3}f(y_+\mid a_0)-c(a_0)
    =
    \bar U+\frac{5}{12}+\frac56-\frac54
    =
    \bar U.
\]
The contract is feasible under limited liability because
$t\geq0$.  It is also a relaxed optimum: under risk neutrality, minimizing
expected compensation is the same as minimizing the agent's expected transfer,
and every relaxed-feasible contract with binding individual rationality has
expected cost $\bar U+c(a_0)$.

However, the agent's expected payoff under this contract is
\[
    t+\frac{20}{3}a^3-5a^2.
\]
Its derivative is $20a^2-10a$, so it is stationary at $a_0=1/2$.  But
\[
    \left(t+\frac{20}{3}(3/4)^3-5(3/4)^2\right)
    -
    \left(t+\frac{20}{3}(1/2)^3-5(1/2)^2\right)
    =
    \frac{5}{12}>0.
\]
The agent strictly prefers $a=3/4$ to the intended action.  Therefore the
first-order approach is not valid for any reservation utility
$\bar U\geq -5/12$, including arbitrarily high reservation utilities.
